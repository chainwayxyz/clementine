1//! # Light Client Proof Verifier
2//! This module implements the light client proof verifier for the bridge circuit.
3//! It includes functions to verify light client proofs and extracting the light client circuit output.
4
5use super::structs::LightClientProof;
6use crate::common::{programs::CURRENT_LIGHT_CLIENT, zkvm::ZkvmGuest};
7use citrea_sov_rollup_interface::zk::light_client_proof::output::LightClientCircuitOutput;
8
9/// Deserializes the light client circuit output from journal bytes.
10fn deserialize_circuit_output(journal: &[u8]) -> LightClientCircuitOutput {
11    borsh::from_slice(journal).expect("Failed to deserialize light client circuit output")
12}
13
14/// Verifies the light client proof and returns the light client circuit output.
15///
16/// Citrea's light client proof is a compressed SP1 proof, verified with native recursion, or
17/// a mock receipt off mainnet, which is not verified.
18pub fn lc_proof_verifier(
19    guest: &impl ZkvmGuest,
20    light_client_proof: LightClientProof,
21) -> LightClientCircuitOutput {
22    let light_client_circuit_output = deserialize_circuit_output(&light_client_proof.lc_journal);
23
24    if light_client_proof.verification_mode.should_verify() {
25        guest.verify(CURRENT_LIGHT_CLIENT.recursion, &light_client_circuit_output);
26    }
27
28    assert_eq!(
29        light_client_circuit_output.light_client_proof_method_id, CURRENT_LIGHT_CLIENT.recursion,
30        "Light client proof method ID does not match the expected light client program key"
31    );
32
33    light_client_circuit_output
34}

36#[cfg(test)]
37mod tests {
38
39    use super::*;
40    use crate::common::{
41        programs::REGTEST, sp1::Receipt, verification_mode::GuestVerificationMode,
42    };
43
44    #[test]
45    fn test_lc_proof_verifier() {
46        let lcp_receipt_bytes = include_bytes!("../../test_data/lcp_receipt.bin");
47        let lcp_receipt: Receipt = borsh::from_slice(lcp_receipt_bytes).unwrap();
48
49        let light_client_proof = LightClientProof::from_receipt(&lcp_receipt);
50
51        let light_client_circuit_output: LightClientCircuitOutput =
52            borsh::from_slice(light_client_proof.lc_journal.as_slice())
53                .expect("Failed to deserialize light client circuit output");
54
55        assert_eq!(
56            light_client_circuit_output.light_client_proof_method_id,
57            REGTEST.light_client.recursion
58        );
59
60        println!("LCP Receipt: {:?}", lcp_receipt.clone());
61
62        assert_eq!(
63            light_client_proof.verification_mode,
64            GuestVerificationMode::SkipForTesting
65        );
66
67        // The guest verifies the proof of the re-encoded output, which must be the journal.
68        assert_eq!(
69            borsh::to_vec(&light_client_circuit_output).unwrap(),
70            light_client_proof.lc_journal
71        );
72
73        let expected_state_root =
74            "87f4f2b4660485ea3ca667033a7fbd077a627946f0b444da101c6cc23a438382";
75        let expected_last_block_hash =
76            "363fd37d55e728d7292f495768540fb7454770a142fe8b44033d836cb0e15d37";
77
78        assert_eq!(
79            hex::encode(light_client_circuit_output.l2_state_root),
80            expected_state_root
81        );
82        assert_eq!(
83            hex::encode(light_client_circuit_output.latest_da_state.block_hash),
84            expected_last_block_hash
85        );
86    }
