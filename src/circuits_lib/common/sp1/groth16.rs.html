1//! SP1 public-value scalars and development Groth16 verification.
2//! Production BitVM key preparation belongs to the host.
3use ark_bn254::{Bn254, Fr};
4use ark_ec::{AffineRepr, CurveGroup};

8/// Decode a canonical big-endian scalar. Never silently reduce an untrusted key hash.
9pub fn scalar_from_bytes(bytes: &[u8; 32]) -> Result<Fr, &'static str> {
10    let scalar = Fr::from_be_bytes_mod_order(bytes);
11    if scalar.into_bigint().to_bytes_be() != bytes {
12        return Err("Non-canonical BN254 scalar");
13    }
14    Ok(scalar)
15}

17/// Fold the fixed first public input into IC[0], leaving exactly one public input.
18/// The program hash must come from the trusted guest build, never from a proof.
19pub fn specialize_vk(
20    mut vk: VerifyingKey<Bn254>,
21    program_hash: &[u8; 32],
22) -> Result<VerifyingKey<Bn254>, &'static str> {
23    if vk.gamma_abc_g1.len() != 3 {
24        return Err("Expected an SP1 key with two public inputs");
25    }
26    let program = scalar_from_bytes(program_hash)?;
27    let constant = (vk.gamma_abc_g1[0].into_group()
28        + vk.gamma_abc_g1[1].mul_bigint(program.into_bigint()))
29    .into_affine();
30    vk.gamma_abc_g1 = vec![constant, vk.gamma_abc_g1[2]];
31    Ok(vk)
32}

34/// SP1's SHA-256 public-values digest; the high three bits are cleared, all 253 others kept.
35pub fn public_input(public_values: &[u8]) -> Fr {
36    Fr::from_be_bytes_mod_order(&sp1_verifier::hash_public_inputs(public_values))
37}
38
39/// SP1's BLAKE3 public-values digest, which the bridge program commits for Bitcoin scripts.
40pub fn public_input_blake3(public_values: &[u8]) -> Fr {
41    Fr::from_be_bytes_mod_order(&sp1_verifier::hash_public_inputs_with_fn(
42        public_values,
43        sp1_verifier::blake3_hash,
44    ))
45}

47/// Deterministic development Groth16 proofs for BitVM integration tests.
48/// They do not prove execution and are never accepted by production builds.
49pub mod development {
50    #[cfg(feature = "use-test-vk")]
51    use {
52        super::{specialize_vk, Bn254, Fr, VerifyingKey},
53        ark_groth16::Proof,
54    };
55
56    /// Envelope prefix that distinguishes development seals from SP1's key selector.
57    pub const SEAL_PREFIX: [u8; 4] = *b"SP1D";
58
59    /// Development seal: the prefix followed by the points in SP1's Gnark layout.
60    pub fn seal(proof: &ark_groth16::Proof<ark_bn254::Bn254>) -> Vec<u8> {
61        let points =
62            crate::bridge_circuit::groth16::CircuitGroth16Proof::new(proof.a, proof.b, proof.c);
63        [SEAL_PREFIX.as_slice(), &points.to_gnark_bytes()].concat()
64    }
65
66    /// Raw points of a development seal.
67    pub fn proof_body(seal: &[u8]) -> Result<&[u8; 256], &'static str> {
68        seal.strip_prefix(SEAL_PREFIX.as_slice())
69            .and_then(|body| body.try_into().ok())
70            .ok_or("Invalid development proof")
71    }
72
73    /// The pinned universal development key, before it is bound to a program.
74    #[cfg(feature = "use-test-vk")]
75    pub fn verifying_key() -> VerifyingKey<Bn254> {
76        use ark_serialize::CanonicalDeserialize;
77        VerifyingKey::deserialize_uncompressed(
78            &include_bytes!("../../bridge_circuit/bin/sp1-development-vk.bin")[..],
79        )
80        .expect("Pinned development key")
81    }
82
83    /// The development key bound to a program, with one public input like the production key.
84    #[cfg(feature = "use-test-vk")]
85    pub fn program_verifying_key(program: &[u8; 32]) -> Result<VerifyingKey<Bn254>, &'static str> {
86        specialize_vk(verifying_key(), program)
87    }
88
89    /// Verifies a development proof for `program` over one public input.
90    #[cfg(feature = "use-test-vk")]
91    pub fn verify(program: &[u8; 32], proof: Proof<Bn254>, public_input: Fr) -> bool {
92        program_verifying_key(program).is_ok_and(|vk| {
93            ark_groth16::Groth16::<Bn254>::verify_proof(
94                &ark_groth16::prepare_verifying_key(&vk),
95                &proof,
96                &[public_input],
97            )
98            .unwrap_or(false)
99        })
100    }
101}

113    #[derive(Clone)]
114    struct TwoInputs {
115        program: Fr,
116        output: Fr,
117    }
118    impl ConstraintSynthesizer<Fr> for TwoInputs {
119        fn generate_constraints(self, cs: ConstraintSystemRef<Fr>) -> Result<(), SynthesisError> {
120            let program = cs.new_input_variable(|| Ok(self.program))?;
121            let output = cs.new_input_variable(|| Ok(self.output))?;
122            let product = cs.new_witness_variable(|| Ok(self.program * self.output))?;
123            cs.enforce_constraint(lc!() + program, lc!() + output, lc!() + product)
124        }
125    }
126    fn bytes(value: Fr) -> [u8; 32] {
127        value.into_bigint().to_bytes_be().try_into().unwrap()
128    }

130    #[test]
131    fn folding_preserves_verification_and_binds_both_inputs() {
132        let mut rng = test_rng();
133        let circuit = TwoInputs {
134            program: Fr::from(17u64),
135            output: public_input(b"bridge output"),
136        };
137        let pk =
138            Groth16::<Bn254>::generate_random_parameters_with_reduction(circuit.clone(), &mut rng)
139                .unwrap();
140        let proof =
141            Groth16::<Bn254>::create_random_proof_with_reduction(circuit.clone(), &pk, &mut rng)
142                .unwrap();
143        assert!(Groth16::<Bn254>::verify_proof(
144            &prepare_verifying_key(&pk.vk),
145            &proof,
146            &[circuit.program, circuit.output]
147        )
148        .unwrap());
149        let folded = specialize_vk(pk.vk.clone(), &bytes(circuit.program)).unwrap();
150        assert_eq!(folded.gamma_abc_g1.len(), 2);
151        let prepared = prepare_verifying_key(&folded);
152        assert!(Groth16::<Bn254>::verify_proof(&prepared, &proof, &[circuit.output]).unwrap());
153        assert!(!Groth16::<Bn254>::verify_proof(
154            &prepared,
155            &proof,
156            &[public_input(b"changed output")]
157        )
158        .unwrap());
159        let wrong_program = specialize_vk(pk.vk, &bytes(Fr::from(18u64))).unwrap();
160        assert!(!Groth16::<Bn254>::verify_proof(
161            &prepare_verifying_key(&wrong_program),
162            &proof,
163            &[circuit.output]
164        )
165        .unwrap());
166        assert!(Groth16::<Bn254>::verify_proof(
167            &prepared,
168            &proof,
169            &[circuit.program, circuit.output]
170        )
171        .is_err());
172    }

174    #[test]
175    fn digests_become_scalars_without_reduction_or_dropped_high_bits() {
176        for i in 0u32..100 {
177            let data = i.to_le_bytes();
178            assert_eq!(
179                bytes(public_input(&data)),
180                sp1_verifier::hash_public_inputs(&data)
181            );
182            assert_eq!(
183                bytes(public_input_blake3(&data)),
184                sp1_verifier::hash_public_inputs_with_fn(&data, sp1_verifier::blake3_hash)
185            );
186        }
187        assert!((0u32..100).any(|i| bytes(public_input(&i.to_le_bytes()))[0] != 0));
188    }

190    #[test]
191    fn development_seals_need_the_prefix_and_exact_length() {
192        let points = crate::bridge_circuit::groth16::CircuitGroth16Proof::new(
193            ark_bn254::G1Affine::generator(),
194            ark_bn254::G2Affine::generator(),
195            ark_bn254::G1Affine::generator(),
196        );
197        let mut seal = development::seal(&points.into());
198        assert_eq!(development::proof_body(&seal), Ok(&points.to_gnark_bytes()));
199        for size in [0, 4, 259] {
200            assert!(development::proof_body(&seal[..size]).is_err());
201        }
202        assert!(development::proof_body(&[seal.as_slice(), &[0]].concat()).is_err());
203        seal[0] ^= 1;
204        assert!(development::proof_body(&seal).is_err());
205    }
