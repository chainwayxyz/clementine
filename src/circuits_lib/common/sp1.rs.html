8#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, BorshSerialize, BorshDeserialize)]

10    pub bytes: Vec<u8>,
11}

13#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, BorshSerialize, BorshDeserialize)]

16}

18/// Compressed SP1 proof, which guests verify with native recursion.
19#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, BorshSerialize, BorshDeserialize)]
20pub struct CompressedProof {
21    /// Program key (`vk.hash_u32()`).
22    pub native_vkey: [u32; 8],
23    /// Bincode-encoded `SP1Proof::Compressed`. Only the host reads this.
24    pub proof: Vec<u8>,
25    /// Bincode-encoded `SP1VerifyingKey`. Only the host reads this.
26    pub vkey: Vec<u8>,
27}
28
29#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, BorshSerialize, BorshDeserialize)]
30pub enum Proof {
31    /// SP1 Groth16 proof over the SHA-256 or BLAKE3 public-values digest.
32    Groth16(Groth16Proof),
33    /// Execution without a proof of the program with key `vk.hash_u32()`. Accepted only where
34    /// mocks are explicitly allowed.
35    Mock { program: [u32; 8] },
36    /// Groth16 proof under the development key, for BitVM integration tests.
37    DevelopmentGroth16 {
38        program: [u8; 32],
39        proof: Groth16Proof,
40    },
41    /// Wire tag 3 matches Citrea; preserve the three preceding discriminants.
42    Compressed(CompressedProof),
43}
44
45#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, BorshSerialize, BorshDeserialize)]
46pub struct Receipt {
47    pub inner: Proof,
48    pub journal: Journal,
49}

51/// Validate v6 metadata before discarding it for Bitcoin's raw Groth16 encoding.
52/// The verification key fixes successful execution, the pinned recursion root,
53/// and a zero proof nonce. These values must never be taken from an untrusted proof.
54pub fn production_proof_body(seal: &[u8]) -> eyre::Result<&[u8; 256]> {
55    use sha2::{Digest, Sha256};
56    eyre::ensure!(seal.len() == 356, "Invalid SP1 v6 proof length");
57    eyre::ensure!(
58        seal[..4] == Sha256::digest(*sp1_verifier::GROTH16_VK_BYTES)[..4],
59        "Invalid SP1 key selector"
60    );
61    eyre::ensure!(seal[4..36] == [0; 32], "SP1 execution did not succeed");
62    eyre::ensure!(
63        seal[36..68] == *sp1_verifier::VK_ROOT_BYTES,
64        "Invalid SP1 recursion root"
65    );
66    eyre::ensure!(
67        seal[68..100] == [0; 32],
68        "Nonzero SP1 proof nonce is not supported by the fixed Bitcoin key"
69    );
70    Ok(seal[100..].try_into().expect("checked length"))
71}
72
73impl Receipt {
74    /// Unproven receipt of an executed guest, accepted only where mocks are allowed.
75    pub fn mock(program: [u32; 8], public_values: Vec<u8>) -> Self {
76        Self {
77            inner: Proof::Mock { program },
78            journal: Journal {
79                bytes: public_values,
80            },
81        }
82    }
83
84    /// Raw points after checking the metadata fixed into the selected Bitcoin key.
85    pub fn groth16_body(&self) -> eyre::Result<&[u8; 256]> {
86        match &self.inner {
87            Proof::Groth16(proof) => production_proof_body(&proof.seal),
88            Proof::DevelopmentGroth16 { proof, .. } => {
89                crate::common::sp1::groth16::development::proof_body(&proof.seal)
90                    .map_err(eyre::Report::msg)
91            }
92            Proof::Mock { .. } | Proof::Compressed(_) => {
93                eyre::bail!("Expected a Groth16 proof");
94            }
95        }
96    }
97
98    /// Groth16 points of a production or development proof, checked to be valid curve points.
99    #[cfg(not(target_os = "zkvm"))]
100    pub fn groth16_proof(&self) -> eyre::Result<ark_groth16::Proof<ark_bn254::Bn254>> {
101        sp1_verifier::load_ark_proof_from_bytes(self.groth16_body()?)
102            .map_err(|_| eyre::eyre!("Invalid SP1 Groth16 points"))
103    }
104}

106/// SP1 guest I/O and native recursion.
107#[cfg(target_os = "zkvm")]
108#[derive(Debug, Clone, Default)]
109pub struct SP1Guest;
110
111#[cfg(target_os = "zkvm")]
112impl crate::common::zkvm::ZkvmGuest for SP1Guest {
113    fn read_from_host<T: BorshDeserialize>(&self) -> T {
114        borsh::from_slice(&sp1_zkvm::io::read_vec()).expect("Failed to deserialize input from host")
115    }
116
117    fn commit<T: BorshSerialize>(&self, item: &T) {
118        sp1_zkvm::io::commit_slice(
119            &borsh::to_vec(item).expect("Serialization to vec is infallible"),
120        );
121    }
122
123    /// Defers verification of a compressed proof of `method_id` (SP1's `vk.hash_u32()`)
124    /// with public values `journal`. The host supplies the proof; real proving fails without it.
125    fn verify<T: BorshSerialize>(&self, method_id: [u32; 8], journal: &T) {
126        use sha2::{Digest, Sha256};
127        let public_values = borsh::to_vec(journal).expect("Serialization to vec is infallible");
128        sp1_zkvm::lib::verify::verify_sp1_proof(&method_id, &Sha256::digest(public_values).into());
129    }
130}

132#[cfg(test)]
133mod tests {
134    use super::*;
135
136    #[test]
137    fn bitcoin_proof_metadata_is_fixed_and_checked_before_compression() {
138        use sha2::{Digest, Sha256};
139        // Point validity is checked by the subsequent Groth16 decoder.
140        let mut seal = vec![0; 356];
141        seal[..4].copy_from_slice(&Sha256::digest(*sp1_verifier::GROTH16_VK_BYTES)[..4]);
142        seal[36..68].copy_from_slice(&*sp1_verifier::VK_ROOT_BYTES);
143        assert!(production_proof_body(&seal).is_ok());
144        for index in [0, 4, 35, 36, 67, 68, 99] {
145            let mut changed = seal.clone();
146            changed[index] ^= 1;
147            assert!(
148                production_proof_body(&changed).is_err(),
149                "accepted changed metadata at {index}"
150            );
151        }
152        for size in [0, 4, 100, 260, 355, 357] {
153            assert!(production_proof_body(&vec![0; size]).is_err());
154        }
155        seal[100..].fill(255);
156        let receipt = Receipt {
157            inner: Proof::Groth16(Groth16Proof { seal }),
158            journal: Journal { bytes: vec![] },
159        };
160        assert!(receipt.groth16_proof().is_err(), "accepted invalid points");
161    }

163    #[test]
164    fn wire_layout_matches_citrea() {
165        let mut receipt = Receipt::mock([7; 8], vec![1, 2, 3]);
166        let wire = borsh::to_vec(&receipt).unwrap();
167        assert_eq!(wire[0], 1);
168        assert_eq!(&wire[1..33], &[7, 0, 0, 0].repeat(8));
169        assert_eq!(&wire[33..], &[3, 0, 0, 0, 1, 2, 3]);
170
171        let mut native_vkey = [0; 8];
172        native_vkey[7] = 7;
173        receipt.inner = Proof::Compressed(CompressedProof {
174            native_vkey,
175            proof: vec![1],
176            vkey: vec![2],
177        });
178        let wire = borsh::to_vec(&receipt).unwrap();
179        assert_eq!(
180            wire[0], 3,
181            "Citrea and Clementine use the same compressed tag"
182        );
183        assert_eq!(&wire[1..5], &[0; 4]);
184        assert_eq!(&wire[29..33], &[7, 0, 0, 0]);
185    }
