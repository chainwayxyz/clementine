1//! SP1 program keys used together on each Bitcoin network.
2//!
3//! `scripts/build_sp1_guests.py` generates the keys in [`super::sp1_programs`]. With `use-test-vk`,
4//! non-mainnet networks use the development bridge guest, which verifies watchtower proofs with
5//! the development Groth16 key. The other guests are the same in both builds.
6use super::{sp1_programs as generated, BuildNetwork, BUILD_NETWORK};
7
8/// Identifier of an SP1 program: the digest of its verifying key, which is eight KoalaBear
9/// field elements. SP1 has two encodings of this digest, because its two verifiers work over
10/// different fields. `scripts/build_sp1_guests.py` generates both from the program ELFs, so no
11/// code converts between them; `generated_key_encodings_match_sp1` checks that they agree.
12#[derive(Debug, Clone, Copy, PartialEq, Eq)]
13pub struct ProgramKey {
14    /// `vk.hash_u32()`: the eight field elements as words, each below 2^31. SP1's native
15    /// recursion (`verify_sp1_proof`) checks compressed proofs over KoalaBear and takes this
16    /// form, so guests use it to verify header chain and light client proofs. Mock receipts,
17    /// circuit outputs and Citrea's `light_client_proof_method_id` also use this form.
18    pub recursion: [u32; 8],
19    /// `vk.bytes32()`: the eight words packed into one big-endian BN254 scalar,
20    /// `sum(recursion[i] << (31 * (7 - i)))`, which fits in 248 bits. SP1's Groth16 proofs are
21    /// over BN254 and take the program as one public input, so Groth16 verification and the
22    /// BitVM key use this form.
23    pub groth16: [u8; 32],
24}
25
26/// SP1 program keys of the guests that one Bitcoin network uses together.
27#[derive(Debug, Clone, Copy, PartialEq, Eq)]
28pub struct NetworkPrograms {
29    pub header_chain: ProgramKey,
30    pub work_only: ProgramKey,
31    pub bridge: ProgramKey,
32    /// Citrea's light client program. Signet uses Citrea devnet.
33    pub light_client: ProgramKey,
34}
35
36/// Replaces the bridge program with its development build when compiled with `use-test-vk`.
37const fn with_development_bridge(
38    programs: NetworkPrograms,
39    development_bridge: ProgramKey,
40) -> NetworkPrograms {
41    if cfg!(feature = "use-test-vk") {
42        NetworkPrograms {
43            bridge: development_bridge,
44            ..programs
45        }
46    } else {
47        programs
48    }
49}
50
51pub const MAINNET: NetworkPrograms = generated::MAINNET;
52pub const TESTNET4: NetworkPrograms =
53    with_development_bridge(generated::TESTNET4, generated::TESTNET4_DEVELOPMENT_BRIDGE);
54pub const SIGNET: NetworkPrograms =
55    with_development_bridge(generated::SIGNET, generated::SIGNET_DEVELOPMENT_BRIDGE);
56pub const REGTEST: NetworkPrograms =
57    with_development_bridge(generated::REGTEST, generated::REGTEST_DEVELOPMENT_BRIDGE);
58
59/// Programs for the network this crate is compiled for, [`BUILD_NETWORK`]. Guests must not
60/// read this table directly: a reference into it can compile the whole table, with the guest's
61/// own key, into the ELF, and that key then depends on itself. Each constant below holds one
62/// program's key.
63const CURRENT: NetworkPrograms = match BUILD_NETWORK {
64    BuildNetwork::Mainnet => MAINNET,
65    BuildNetwork::Testnet4 => TESTNET4,
66    BuildNetwork::Signet => SIGNET,
67    BuildNetwork::Regtest => REGTEST,
68};
69/// The header chain program of [`BUILD_NETWORK`].
70pub const CURRENT_HEADER_CHAIN: ProgramKey = CURRENT.header_chain;
71/// The work-only program of [`BUILD_NETWORK`].
72pub const CURRENT_WORK_ONLY: ProgramKey = CURRENT.work_only;
73/// Citrea's light client program of [`BUILD_NETWORK`].
74pub const CURRENT_LIGHT_CLIENT: ProgramKey = CURRENT.light_client;
75
76impl NetworkPrograms {
77    /// Programs for a runtime network, or `None` if Clementine does not support it.
78    pub fn for_network(network: bitcoin::Network) -> Option<&'static Self> {
79        match network {
80            bitcoin::Network::Bitcoin => Some(&MAINNET),
81            bitcoin::Network::Testnet4 => Some(&TESTNET4),
82            bitcoin::Network::Signet => Some(&SIGNET),
83            bitcoin::Network::Regtest => Some(&REGTEST),
84            _ => None,
85        }
86    }
87}

89#[cfg(test)]
90mod tests {
91    use super::*;
92
93    /// Check the generated native words against SP1's field packing, including every
94    /// network's Citrea key (whose ELF is supplied by Citrea rather than built here).
95    #[test]
96    fn generated_key_encodings_match_sp1() {
97        use slop_algebra::{AbstractField, PrimeField};
98
99        let networks = [MAINNET, TESTNET4, SIGNET, REGTEST];
100        let development = [
101            generated::TESTNET4_DEVELOPMENT_BRIDGE,
102            generated::SIGNET_DEVELOPMENT_BRIDGE,
103            generated::REGTEST_DEVELOPMENT_BRIDGE,
104        ];
105        for key in networks
106            .into_iter()
107            .flat_map(|network| {
108                [
109                    network.header_chain,
110                    network.work_only,
111                    network.bridge,
112                    network.light_client,
113                ]
114            })
115            .chain(development)
116        {
117            assert!(key.recursion.iter().all(|word| *word < 0x7f00_0001));
118            let fields = key.recursion.map(AbstractField::from_canonical_u32);
119            let expected = sp1_hypercube::koalabears_to_bn254(&fields)
120                .as_canonical_biguint()
121                .to_bytes_be();
122            let mut padded = [0; 32];
123            padded[32 - expected.len()..].copy_from_slice(&expected);
124            assert_eq!(key.groth16, padded);
125        }
126    }
