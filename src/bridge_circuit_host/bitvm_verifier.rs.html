1//! BitVM verifying keys of bridge programs, prepared for Bitcoin's single-public-input
2//! verifier from SP1's production Groth16 key or the development key.

9/// BitVM verifying key of a bridge program: the development key with `use-test-vk`,
10/// SP1's Groth16 key otherwise. The program is folded in, leaving one public input.
11pub fn get_verifying_key(program: &[u8; 32]) -> eyre::Result<VerifyingKey<Bn254>> {
12    #[cfg(feature = "use-test-vk")]
13    let key = circuits_lib::common::sp1::groth16::development::program_verifying_key(program);
14    #[cfg(not(feature = "use-test-vk"))]
15    let key = program_verifying_key(program);
16    key.map_err(eyre::Report::msg)
17}
18
19/// Verifies a bridge proof as Bitcoin does, with the program's BitVM key and one public input.
20pub fn verify_bitvm(
21    program: &[u8; 32],
22    proof: &Proof<Bn254>,
23    public_input: Fr,
24) -> eyre::Result<()> {
25    let key = get_verifying_key(program)?;
26    let valid =
27        Groth16::<Bn254>::verify_proof(&prepare_verifying_key(&key), proof, &[public_input])
28            .map_err(|e| eyre::eyre!("BitVM Groth16 verification failed: {e}"))?;
29    eyre::ensure!(valid, "BitVM Groth16 verification failed");
30    Ok(())
31}

33/// Fix v6 metadata to successful execution, the trusted recursion root, and zero nonce.
34pub fn fix_sp1_metadata(mut vk: VerifyingKey<Bn254>) -> Result<VerifyingKey<Bn254>, &'static str> {
35    if vk.gamma_abc_g1.len() != 6 {
36        return Err("Expected five SP1 v6 public inputs");
37    }
38    let root = Fr::from_be_bytes_mod_order(&*sp1_verifier::VK_ROOT_BYTES);
39    vk.gamma_abc_g1[0] = (vk.gamma_abc_g1[0].into_group()
40        + vk.gamma_abc_g1[4].mul_bigint(root.into_bigint()))
41    .into_affine();
42    // Inputs 2 (exit code) and 4 (nonce) are zero; input 3 is the pinned root.
43    vk.gamma_abc_g1.truncate(3);
44    Ok(vk)
45}
46
47/// Load SP1's pinned universal Groth16 key and bind it to a program.
48pub fn program_verifying_key(program_hash: &[u8; 32]) -> Result<VerifyingKey<Bn254>, &'static str> {
49    let vk =
50        sp1_verifier::load_ark_groth16_verifying_key_from_bytes(*sp1_verifier::GROTH16_VK_BYTES)
51            .map_err(|_| "Invalid SP1 Groth16 key")?;
52    specialize_vk(fix_sp1_metadata(vk)?, program_hash)
53}

70    #[derive(Clone)]
71    struct FiveInputs([Fr; 5]);
72    impl ConstraintSynthesizer<Fr> for FiveInputs {
73        fn generate_constraints(self, cs: ConstraintSystemRef<Fr>) -> Result<(), SynthesisError> {
74            for value in self.0 {
75                let input = cs.new_input_variable(|| Ok(value))?;
76                let square = cs.new_witness_variable(|| Ok(value * value))?;
77                cs.enforce_constraint(lc!() + input, lc!() + input, lc!() + square)?;
78            }
79            Ok(())
80        }
81    }

83    #[test]
84    fn v6_metadata_folding_rejects_failed_exit_wrong_root_and_nonzero_nonce() {
85        let mut rng = test_rng();
86        let inputs = [
87            Fr::from(17u64),
88            public_input(b"output"),
89            Fr::from(0u64),
90            Fr::from_be_bytes_mod_order(&*sp1_verifier::VK_ROOT_BYTES),
91            Fr::from(0u64),
92        ];
93        let pk = Groth16::<Bn254>::generate_random_parameters_with_reduction(
94            FiveInputs(inputs),
95            &mut rng,
96        )
97        .unwrap();
98        let key =
99            specialize_vk(fix_sp1_metadata(pk.vk.clone()).unwrap(), &bytes(inputs[0])).unwrap();
100        let prepared = prepare_verifying_key(&key);
101        assert_eq!(key.gamma_abc_g1.len(), 2);
102        let proof =
103            Groth16::<Bn254>::create_random_proof_with_reduction(FiveInputs(inputs), &pk, &mut rng)
104                .unwrap();
105        assert!(Groth16::<Bn254>::verify_proof(&prepared, &proof, &[inputs[1]]).unwrap());
106        for index in [0, 1, 2, 3, 4] {
107            let mut changed = inputs;
108            changed[index] += Fr::from(1u64);
109            let proof = Groth16::<Bn254>::create_random_proof_with_reduction(
110                FiveInputs(changed),
111                &pk,
112                &mut rng,
113            )
114            .unwrap();
115            assert!(
116                !Groth16::<Bn254>::verify_proof(&prepared, &proof, &[inputs[1]]).unwrap(),
117                "input {index} was not bound"
118            );
119        }
120    }
