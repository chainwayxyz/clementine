1//! Guest ELFs embedded in the host, each paired with its program key from
2//! [`circuits_lib::common::programs`].
3use crate::sp1::ProofKind;
4use circuits_lib::common::programs::{self, ProgramKey};
5use std::fmt;
6
7/// A trusted guest ELF, the program key generated from it, and the proof its consumer needs.
8#[derive(Clone, Copy)]
9pub struct Guest {
10    pub elf: &'static [u8],
11    pub key: ProgramKey,
12    pub kind: ProofKind,
13}
14
15impl fmt::Debug for Guest {
16    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
17        f.debug_struct("Guest")
18            .field("key", &self.key)
19            .field("kind", &self.kind)
20            .field("elf_len", &self.elf.len())
21            .finish()
22    }
23}
24
25/// Guests that one Bitcoin network uses together, and Citrea's light client program.
26#[derive(Debug)]
27pub struct NetworkGuests {
28    pub header_chain: Guest,
29    pub work_only: Guest,
30    pub bridge: Guest,
31    pub light_client: ProgramKey,
32}
33
34macro_rules! elf {
35    ($name:expr) => {
36        include_bytes!(concat!("../../sp1-circuits/elfs/", $name, ".elf"))
37    };
38}
39
40/// Guests of a test network. With `use-test-vk`, the bridge guest is the development build,
41/// whose key the network's programs also select.
42macro_rules! test_network_guests {
43    ($network:literal, $programs:expr) => {
44        NetworkGuests {
45            header_chain: Guest {
46                elf: elf!(concat!($network, "-header-chain")),
47                key: $programs.header_chain,
48                kind: ProofKind::Compressed,
49            },
50            work_only: Guest {
51                elf: elf!(concat!($network, "-work-only")),
52                key: $programs.work_only,
53                kind: ProofKind::Groth16,
54            },
55            bridge: Guest {
56                #[cfg(not(feature = "use-test-vk"))]
57                elf: elf!(concat!($network, "-bridge-circuit")),
58                #[cfg(feature = "use-test-vk")]
59                elf: elf!(concat!("test-", $network, "-bridge-circuit")),
60                key: $programs.bridge,
61                kind: ProofKind::BridgeGroth16,
62            },
63            light_client: $programs.light_client,
64        }
65    };
66}
67
68pub static MAINNET: NetworkGuests = NetworkGuests {
69    header_chain: Guest {
70        elf: elf!("mainnet-header-chain"),
71        key: programs::MAINNET.header_chain,
72        kind: ProofKind::Compressed,
73    },
74    work_only: Guest {
75        elf: elf!("mainnet-work-only"),
76        key: programs::MAINNET.work_only,
77        kind: ProofKind::Groth16,
78    },
79    bridge: Guest {
80        elf: elf!("mainnet-bridge-circuit"),
81        key: programs::MAINNET.bridge,
82        kind: ProofKind::BridgeGroth16,
83    },
84    light_client: programs::MAINNET.light_client,
85};
86pub static TESTNET4: NetworkGuests = test_network_guests!("testnet4", programs::TESTNET4);
87pub static SIGNET: NetworkGuests = test_network_guests!("signet", programs::SIGNET);
88pub static REGTEST: NetworkGuests = test_network_guests!("regtest", programs::REGTEST);
89
90impl NetworkGuests {
91    /// Guests for a runtime network, or `None` if Clementine does not support it.
92    pub fn for_network(network: bitcoin::Network) -> Option<&'static Self> {
93        match network {
94            bitcoin::Network::Bitcoin => Some(&MAINNET),
95            bitcoin::Network::Testnet4 => Some(&TESTNET4),
96            bitcoin::Network::Signet => Some(&SIGNET),
97            bitcoin::Network::Regtest => Some(&REGTEST),
98            _ => None,
99        }
100    }
