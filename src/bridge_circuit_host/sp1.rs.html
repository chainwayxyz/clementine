14/// Development programs require mock execution; production programs require real proofs.
15pub fn check_programs(backend: Backend) -> Result<Backend> {
16    eyre::ensure!(
17        (backend == Backend::Mock) == is_test_vk(),
18        "SP1_PROVER {backend:?} does not match the programs; use mock with use-test-vk, or cpu/network without it"
19    );
20    Ok(backend)
21}

23/// The proof that [`prove`] produces for a guest, fixed by the guest's consumer.
24#[derive(Debug, Clone, Copy, PartialEq, Eq)]
25pub enum ProofKind {
26    /// Compressed STARK, which other guests verify with native recursion.
27    Compressed,
28    /// Groth16 over the SHA-256 public-values digest, which the bridge guest verifies.
29    Groth16,
30    /// Groth16 over the BLAKE3 public-values digest, which BitVM verifies.
31    BridgeGroth16,
32}

34/// Guest input: the Borsh-encoded circuit input and the native recursion witnesses of its
35/// assumptions, in the order the guest verifies them.
36pub struct GuestInput {
37    stdin: SP1Stdin,
38    has_mock_assumption: bool,
39}
40
41impl GuestInput {
42    pub fn new(input: &impl borsh::BorshSerialize) -> Result<Self> {
43        let mut stdin = SP1Stdin::new();
44        stdin.write_vec(borsh::to_vec(input)?);
45        Ok(Self {
46            stdin,
47            has_mock_assumption: false,
48        })
49    }
50
51    /// Adds a proof that the guest verifies with `ZkvmGuest::verify`. Mock execution does not
52    /// check these, so mock receipts need no witness.
53    pub fn with_assumption(mut self, receipt: &Receipt) -> Result<Self> {
54        ensure_assumption(receipt)?;
55        match &receipt.inner {
56            Proof::Compressed(proof) => write_native_witness(&mut self.stdin, proof)?,
57            _ => self.has_mock_assumption = true,
58        }
59        Ok(self)
60    }
61}

63/// Ensures that guests can use `receipt` as an assumption. Guests verify only compressed
64/// proofs, with native recursion; mock receipts stand in for them in tests.
65pub fn ensure_assumption(receipt: &Receipt) -> Result<()> {
66    eyre::ensure!(
67        matches!(receipt.inner, Proof::Compressed(_) | Proof::Mock { .. }),
68        "Assumptions must be compressed or mock receipts"
69    );
70    Ok(())
71}

73fn write_native_witness(stdin: &mut SP1Stdin, proof: &CompressedProof) -> Result<()> {
74    let vk: sp1_sdk::SP1VerifyingKey = bincode::deserialize(&proof.vkey)?;
75    eyre::ensure!(
76        vk.hash_u32() == proof.native_vkey,
77        "Compressed proof key mismatch"
78    );
79    let sp1_sdk::SP1Proof::Compressed(inner) = bincode::deserialize(&proof.proof)? else {
80        eyre::bail!("Expected a compressed SP1 proof");
81    };
82    stdin.write_proof(*inner, vk.vk);
83    Ok(())
84}

92/// Verifies a real SP1 Groth16 or compressed proof of `program` with SP1's verifiers, which
93/// accept either public-values digest. With `allow_mock`, it also accepts mock receipts and
94/// development Groth16 proofs of `program`.
95pub async fn verify(receipt: &Receipt, program: ProgramKey, allow_mock: bool) -> Result<()> {
96    let public_values = &receipt.journal.bytes;
97    match &receipt.inner {
98        Proof::Groth16(proof) => sp1_verifier::Groth16Verifier::verify(
99            &proof.seal,
100            public_values,
101            &format!("0x{}", hex::encode(program.groth16)),
102            &sp1_verifier::GROTH16_VK_BYTES,
103        )
104        .map_err(|e| eyre!("SP1 Groth16 verification failed: {e:?}")),
105        Proof::Compressed(proof) => {
106            let vk: SP1VerifyingKey = bincode::deserialize(&proof.vkey)?;
107            eyre::ensure!(
108                vk.hash_u32() == program.recursion && proof.native_vkey == program.recursion,
109                "Compressed SP1 program mismatch"
110            );
111            sp1_host::verify_compressed(bincode::deserialize(&proof.proof)?, public_values, vk)
112                .await
113        }
114        Proof::Mock { program: executed } => {
115            eyre::ensure!(
116                allow_mock && *executed == program.recursion,
117                "Mock SP1 receipt is not allowed or has the wrong program"
118            );
119            Ok(())
120        }
121        Proof::DevelopmentGroth16 {
122            program: proven, ..
123        } => {
124            eyre::ensure!(
125                allow_mock && *proven == program.groth16,
126                "Development proof is not allowed or has the wrong program"
127            );
128            #[cfg(feature = "use-test-vk")]
129            {
130                let proof = receipt.groth16_proof()?;
131                let digests = [
132                    bridge_sp1::public_input(public_values),
133                    bridge_sp1::public_input_blake3(public_values),
134                ];
135                eyre::ensure!(
136                    digests.into_iter().any(|public_input| {
137                        bridge_sp1::development::verify(
138                            &program.groth16,
139                            proof.clone(),
140                            public_input,
141                        )
142                    }),
143                    "Invalid development proof"
144                );
145                Ok(())
146            }
147            #[cfg(not(feature = "use-test-vk"))]
148            {
149                eyre::bail!("Development verification was not compiled")
150            }
151        }
152    }
153}

155/// Proves a trusted guest as its [`ProofKind`] and verifies the result against the guest's
156/// program key.
157///
158/// With [`Backend::Mock`], the guest is only executed. Groth16 kinds then carry a development
159/// Groth16 proof when compiled with `use-test-vk`, for BitVM integration tests.
160pub async fn prove(guest: &Guest, input: GuestInput, backend: Backend) -> Result<Receipt> {
161    let kind = guest.kind;
162    eyre::ensure!(
163        backend == Backend::Mock || !input.has_mock_assumption,
164        "Real SP1 proving needs compressed assumptions, not mock receipts"
165    );
166    let host = Host::get(guest.elf, backend).await?;
167    let key = guest.key;
168    eyre::ensure!(
169        host.program_id()? == key.groth16,
170        "Guest ELF does not match its program key"
171    );
172    let receipt = match backend {
173        Backend::Mock => {
174            let mut receipt = Receipt::mock(key.recursion, host.execute(input.stdin).await?);
175            if kind != ProofKind::Compressed && is_test_vk() {
176                receipt.inner = development_groth16(&key.groth16, &receipt.journal.bytes, kind)?;
177            }
178            receipt
179        }
180        Backend::Cpu | Backend::Network => prove_real(&host, input.stdin, kind).await?,
181    };
182    verify(&receipt, key, backend == Backend::Mock).await?;
183    Ok(receipt)
184}

186async fn prove_real(host: &Host, stdin: SP1Stdin, kind: ProofKind) -> Result<Receipt> {
187    let mode = match kind {
188        ProofKind::Compressed => SP1ProofMode::Compressed,
189        ProofKind::Groth16 | ProofKind::BridgeGroth16 => SP1ProofMode::Groth16,
190    };
191    let proof = host
192        .prove(stdin, mode, [0; 4], kind == ProofKind::BridgeGroth16)
193        .await?;
194    let vk = host.verifying_key();
195    Ok(Receipt {
196        inner: match kind {
197            ProofKind::Compressed => Proof::Compressed(CompressedProof {
198                native_vkey: vk.hash_u32(),
199                proof: bincode::serialize(&proof.proof)?,
200                vkey: bincode::serialize(vk)?,
201            }),
202            ProofKind::Groth16 | ProofKind::BridgeGroth16 => Proof::Groth16(Groth16Proof {
203                seal: groth16_seal(&proof.proof)?,
204            }),
205        },
206        journal: Journal {
207            bytes: proof.public_values.to_vec(),
208        },
209    })
210}

212/// Encodes a Groth16 proof as `vkey_hash[..4] || proof`, like
213/// `SP1ProofWithPublicValues::bytes`. The proof can come from the prover network, so a
214/// malformed proof is an error instead of the SDK's panic.
215fn groth16_seal(proof: &sp1_sdk::SP1Proof) -> Result<Vec<u8>> {
216    let sp1_sdk::SP1Proof::Groth16(proof) = proof else {
217        eyre::bail!("Expected an SP1 Groth16 proof");
218    };
219    let encoded = hex::decode(&proof.encoded_proof)
220        .map_err(|e| eyre!("Invalid SP1 Groth16 proof encoding: {e}"))?;
221    Ok([&proof.groth16_vkey_hash[..4], encoded.as_slice()].concat())
222}

224/// Development Groth16 proof over the public values, for BitVM integration tests. It is not
225/// a proof of execution.
226fn development_groth16(program: &[u8; 32], public_values: &[u8], kind: ProofKind) -> Result<Proof> {
227    #[cfg(feature = "use-test-vk")]
228    {
229        let public_input = match kind {
230            ProofKind::BridgeGroth16 => bridge_sp1::public_input_blake3(public_values),
231            ProofKind::Groth16 | ProofKind::Compressed => bridge_sp1::public_input(public_values),
232        };
233        let proof = circuits_lib::bridge_circuit::sp1_test_prover::prove(program, public_input)
234            .map_err(eyre::Report::msg)?;
235        Ok(Proof::DevelopmentGroth16 {
236            program: *program,
237            proof: Groth16Proof {
238                seal: bridge_sp1::development::seal(&proof),
239            },
240        })
241    }
242    #[cfg(not(feature = "use-test-vk"))]
243    {
244        let _ = (program, public_values, kind);
245        Err(eyre!("Development Groth16 proofs require use-test-vk"))
246    }
247}
