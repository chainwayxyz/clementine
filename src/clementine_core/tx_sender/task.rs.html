1//! # Transaction Sender Task
2//!
3//! This module provides the [`Task`] implementation for the [`TxSender`].
4//!
5//! This task will fetch block events from [`Bitcoin Syncer`](crate::bitcoin_syncer)
6//! to keep the consumer cursor and tip height up to date. Confirmations are
7//! reconciled against the canonical chain once, immediately before sending.
8//! Transactions are picked from the database and sent to the Bitcoin network if
9//! a transaction is in queue and not in the [`Bitcoin Syncer`](crate::bitcoin_syncer)
10//! database.
11
12use super::TxSender;
13use crate::errors::ResultExt;
14use crate::task::{IgnoreError, TaskVariant, WithDelay};
15use crate::{
16    bitcoin_syncer::BitcoinSyncerEvent,
17    database::Database,
18    errors::BridgeError,
19    task::{IntoTask, Task, TaskExt},
20};
21use std::time::Duration;
22use tonic::async_trait;
23
24const POLL_DELAY: Duration = if cfg!(test) {
25    Duration::from_millis(250)
26} else {
27    Duration::from_secs(30)
28};
29
30#[derive(Debug)]
31pub struct TxSenderTask {
32    db: Database,
33    /// `None` until the first run seeds it from this consumer's persisted syncer cursor. Defaulting
34    /// to 0 would make every activation condition look unmet (a timelock is satisfied once
35    /// `seen_height + timelock <= tip`), so after a restart no timelock-gated transaction could be
36    /// sent until the next block produced a `NewBlock` event.
37    current_tip_height: Option<u32>,
38    last_processed_tip_height: u32,
39    inner: TxSender,
40}
41
42impl TxSenderTask {
43    /// Returns the current tip height, seeding it from the database on first use.
44    async fn tip_height(&mut self) -> Result<u32, BridgeError> {
45        match self.current_tip_height {
46            Some(height) => Ok(height),
47            None => {
48                let height = self
49                    .db
50                    .get_last_processed_event_block_height(None, &self.inner.btc_syncer_consumer_id)
51                    .await?
52                    .unwrap_or(0);
53                self.current_tip_height = Some(height);
54                Ok(height)
55            }
56        }
57    }
58}
59
60#[async_trait]
61impl Task for TxSenderTask {
62    type Output = bool;
63    const VARIANT: TaskVariant = TaskVariant::TxSender;
64
65    #[tracing::instrument(skip(self), name = "tx_sender_task")]
66    async fn run_once(&mut self) -> std::result::Result<Self::Output, BridgeError> {
67        let mut dbtx = self.db.begin_transaction().await.map_to_eyre()?;
68
69        // Events only move the cursor and tip. Reconcile once below, before send.
70
71        if let Some(event) = self
72            .db
73            .fetch_next_bitcoin_syncer_evt(&mut dbtx, &self.inner.btc_syncer_consumer_id)
74            .await?
75        {
76            tracing::debug!("Received Bitcoin syncer event: {:?}", event);
77
78            match event {
79                BitcoinSyncerEvent::NewBlock(block_id) => {
80                    let block_height = self
81                        .db
82                        .get_block_info_from_id(Some(&mut dbtx), block_id)
83                        .await?
84                        .ok_or(eyre::eyre!("Block not found in TxSenderTask"))?
85                        .1;
86                    tracing::debug!(
87                        height = block_height,
88                        block_id = %block_id,
89                        "New block"
90                    );
91                    dbtx.commit().await?;
92                    self.current_tip_height = Some(block_height);
93                }
94                BitcoinSyncerEvent::ReorgedBlock(block_id) => {
95                    let height = self
96                        .db
97                        .get_block_info_from_id(Some(&mut dbtx), block_id)
98                        .await?
99                        .ok_or(eyre::eyre!("Block not found in TxSenderTask"))?
100                        .1;
101                    tracing::debug!(
102                        height = height,
103                        block_id = %block_id,
104                        "Reorged block"
105                    );
106                    dbtx.commit().await?;
107                }
108            }
109
110            // More events may be queued; drain them before sending.
111            return Ok(true);
112        }
113
114        self.db.sync_transaction_confirmations(&mut dbtx).await?;
115        dbtx.commit().await?;
116
117        let current_tip_height = self.tip_height().await?;
118
119        tracing::debug!("TXSENDER: Getting fee rate");
120        let fee_rate_result = self.inner.get_fee_rate().await;
121        tracing::debug!("TXSENDER: Fee rate result: {:?}", fee_rate_result);
122        let fee_rate = fee_rate_result?;
123
124        self.inner
125            .try_to_send_unconfirmed_txs(
126                fee_rate,
127                current_tip_height,
128                self.last_processed_tip_height != current_tip_height,
129            )
130            .await?;
131        self.last_processed_tip_height = current_tip_height;
132
133        Ok(false)
134    }
135}

137#[cfg(test)]
138impl TxSenderTask {
139    pub fn new_for_test(inner: TxSender) -> Self {
140        Self {
141            db: inner.db.clone(),
142            current_tip_height: None,
143            last_processed_tip_height: 0,
144            inner,
145        }
146    }
147}
148
149impl IntoTask for TxSender {
150    type Task = WithDelay<IgnoreError<TxSenderTask>>;
151
152    fn into_task(self) -> Self::Task {
153        TxSenderTask {
154            db: self.db.clone(),
155            current_tip_height: None,
156            last_processed_tip_height: 0,
157            inner: self,
158        }
159        .ignore_error()
160        .with_delay(POLL_DELAY)
161    }
162}
