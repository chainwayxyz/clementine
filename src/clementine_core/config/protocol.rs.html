26/// A pre-defined paramset name that can be converted into a
27/// [`ProtocolParamset`] reference.
28///
29/// See: [`REGTEST_PARAMSET`]
30pub enum ProtocolParamsetName {
31    // Pre-defined paramsets
32    Regtest,
33}
34
35impl FromStr for ProtocolParamsetName {
36    type Err = BridgeError;
37
38    fn from_str(s: &str) -> Result<Self, Self::Err> {
39        match s {
40            "regtest" => Ok(ProtocolParamsetName::Regtest),
41            _ => Err(BridgeError::ConfigError(format!(
42                "Unknown paramset name: {s}"
43            ))),
44        }
45    }
46}
47
48impl Display for ProtocolParamsetName {
49    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
50        match self {
51            ProtocolParamsetName::Regtest => write!(f, "regtest"),
52        }
53    }
54}
55
56impl From<ProtocolParamsetName> for &'static ProtocolParamset {
57    fn from(name: ProtocolParamsetName) -> Self {
58        match name {
59            ProtocolParamsetName::Regtest => &REGTEST_PARAMSET,
60        }
61    }
62}

64#[derive(Debug, Clone, PartialEq, Eq, Hash, Deserialize, Serialize)]
65/// Protocol parameters that affect the transactions in the contract (which also
66/// change the pre-calculated txids and sighashes).
67///
68/// These parameters are used when generating the transactions and changing them
69/// will break compatibility between actors, making deposits impossible.  A
70/// paramset is chosen by the actor by choosing a ParamsetName inside the
71/// [`crate::config::BridgeConfig`].
72pub struct ProtocolParamset {
73    /// Bitcoin network to work on (mainnet, testnet, regtest).
74    pub network: Network,
75    /// Number of round transactions that the operator will create.
76    pub num_round_txs: usize,
77    /// Number of kickoff UTXOs per round transaction.
78    pub num_kickoffs_per_round: usize,
79    /// Number of kickoffs that are signed per round and deposit.
80    /// There are num_kickoffs_per_round utxo's, but only num_signed_kickoffs are signed.
81    pub num_signed_kickoffs: usize,
82    /// Bridge deposit amount that users can deposit.
83    pub bridge_amount: Amount,
84    /// Amount allocated for each kickoff UTXO.
85    pub kickoff_amount: Amount,
86    /// Amount allocated for operator challenge transactions.
87    pub operator_challenge_amount: Amount,
88    /// Collateral funding amount for operators used to fund the round transaction chain.
89    pub collateral_funding_amount: Amount,
90    /// Length of the blockhash commitment in kickoff transactions.
91    pub kickoff_blockhash_commit_length: u32,
92    /// Total number of bytes of a watchtower challenge.
93    pub watchtower_challenge_bytes: usize,
94    /// Winternitz derivation log_d (shared for all WOTS commitments)
95    /// Currently used in statics and thus cannot be different from [`WINTERNITZ_LOG_D`].
96    pub winternitz_log_d: u32,
97    /// Number of blocks after which user can take deposit back if deposit request fails.
98    pub user_takes_after: u16,
99    /// Number of blocks for operator challenge timeout timelock (currently BLOCKS_PER_WEEK)
100    pub operator_challenge_timeout_timelock: u16,
101    /// Number of blocks for operator challenge NACK timelock (currently BLOCKS_PER_WEEK * 3)
102    pub operator_challenge_nack_timelock: u16,
103    /// Number of blocks for disprove timeout timelock (currently BLOCKS_PER_WEEK * 5)
104    pub disprove_timeout_timelock: u16,
105    /// Number of blocks for assert timeout timelock (currently BLOCKS_PER_WEEK * 4)
106    pub assert_timeout_timelock: u16,
107    /// Number of blocks for latest blockhash timeout timelock (currently BLOCKS_PER_WEEK * 2.5)
108    pub latest_blockhash_timeout_timelock: u16,
109    /// Number of blocks for operator reimburse timelock (currently BLOCKS_PER_DAY * 2)
110    /// Timelocks operator from sending the next Round Tx after the Ready to Reimburse Tx.
111    pub operator_reimburse_timelock: u16,
112    /// Number of blocks for watchtower challenge timeout timelock (currently BLOCKS_PER_WEEK * 2)
113    pub watchtower_challenge_timeout_timelock: u16,
114    /// Amount of depth a block should have from the current head to be considered finalized
115    /// Also means finality_confirmations, how many confirmations are needed for a block to be considered finalized
116    /// The chain tip has 1 confirmation. Minimum value should be 1.
117    pub finality_depth: u32,
118    /// start height to sync the chain from, i.e. the height bridge was deployed
119    pub start_height: u32,
120    /// Genesis height to sync the header chain proofs from. Can be above `start_height`.
121    /// Citrea light client proofs, deposits and withdrawals are synced from the larger of
122    /// the two, because payouts below genesis cannot be proven. Must be above Citrea's
123    /// first light client proof height. See docs/sp1.md#deployment.
124    pub genesis_height: u32,
125    /// Genesis chain state hash
126    pub genesis_chain_state_hash: [u8; 32],
127    /// Denotes if the bridge is non-standard, i.e. uses 0 sat outputs for round tx (except collateral) and kickoff outputs
128    pub bridge_nonstandard: bool,
129}

131impl ProtocolParamset {
132    pub fn from_toml_file(path: &Path) -> Result<Self, BridgeError> {
133        let contents = fs::read_to_string(path).wrap_err("Failed to read config file")?;
134
135        let paramset: Self = toml::from_str(&contents).wrap_err("Failed to parse TOML")?;
136        if paramset.finality_depth < 1 {
137            return Err(BridgeError::ConfigError(
138                "Finality depth must be at least 1".to_string(),
139            ));
140        }
141
142        Ok(paramset)
143    }
144    pub fn from_env() -> Result<Self, BridgeError> {
145        let config = ProtocolParamset {
146            network: read_string_from_env_then_parse::<Network>("NETWORK")?,
147            num_round_txs: read_string_from_env_then_parse::<usize>("NUM_ROUND_TXS")?,
148            num_kickoffs_per_round: read_string_from_env_then_parse::<usize>(
149                "NUM_KICKOFFS_PER_ROUND",
150            )?,
151            num_signed_kickoffs: read_string_from_env_then_parse::<usize>("NUM_SIGNED_KICKOFFS")?,
152            bridge_amount: Amount::from_sat(read_string_from_env_then_parse::<u64>(
153                "BRIDGE_AMOUNT",
154            )?),
155            kickoff_amount: Amount::from_sat(read_string_from_env_then_parse::<u64>(
156                "KICKOFF_AMOUNT",
157            )?),
158            operator_challenge_amount: Amount::from_sat(read_string_from_env_then_parse::<u64>(
159                "OPERATOR_CHALLENGE_AMOUNT",
160            )?),
161            collateral_funding_amount: Amount::from_sat(read_string_from_env_then_parse::<u64>(
162                "COLLATERAL_FUNDING_AMOUNT",
163            )?),
164            kickoff_blockhash_commit_length: read_string_from_env_then_parse::<u32>(
165                "KICKOFF_BLOCKHASH_COMMIT_LENGTH",
166            )?,
167            watchtower_challenge_bytes: read_string_from_env_then_parse::<usize>(
168                "WATCHTOWER_CHALLENGE_BYTES",
169            )?,
170            winternitz_log_d: read_string_from_env_then_parse::<u32>("WINTERNITZ_LOG_D")?,
171            user_takes_after: read_string_from_env_then_parse::<u16>("USER_TAKES_AFTER")?,
172            operator_challenge_timeout_timelock: read_string_from_env_then_parse::<u16>(
173                "OPERATOR_CHALLENGE_TIMEOUT_TIMELOCK",
174            )?,
175            operator_challenge_nack_timelock: read_string_from_env_then_parse::<u16>(
176                "OPERATOR_CHALLENGE_NACK_TIMELOCK",
177            )?,
178            disprove_timeout_timelock: read_string_from_env_then_parse::<u16>(
179                "DISPROVE_TIMEOUT_TIMELOCK",
180            )?,
181            assert_timeout_timelock: read_string_from_env_then_parse::<u16>(
182                "ASSERT_TIMEOUT_TIMELOCK",
183            )?,
184            operator_reimburse_timelock: read_string_from_env_then_parse::<u16>(
185                "OPERATOR_REIMBURSE_TIMELOCK",
186            )?,
187            watchtower_challenge_timeout_timelock: read_string_from_env_then_parse::<u16>(
188                "WATCHTOWER_CHALLENGE_TIMEOUT_TIMELOCK",
189            )?,
190            finality_depth: read_string_from_env_then_parse::<u32>("FINALITY_DEPTH")?,
191            start_height: read_string_from_env_then_parse::<u32>("START_HEIGHT")?,
192            genesis_height: read_string_from_env_then_parse::<u32>("GENESIS_HEIGHT")?,
193            genesis_chain_state_hash: convert_hex_string_to_bytes(
194                &read_string_from_env_then_parse::<String>("GENESIS_CHAIN_STATE_HASH")?,
195            )?,
196            latest_blockhash_timeout_timelock: read_string_from_env_then_parse::<u16>(
197                "LATEST_BLOCKHASH_TIMEOUT_TIMELOCK",
198            )?,
199            bridge_nonstandard: read_string_from_env_then_parse::<bool>("BRIDGE_NONSTANDARD")?,
200        };
201
202        if config.finality_depth < 1 {
203            return Err(BridgeError::ConfigError(
204                "Finality depth must be at least 1".to_string(),
205            ));
206        }
207
208        Ok(config)
209    }

211    pub fn default_utxo_amount(&self) -> Amount {
212        if self.bridge_nonstandard {
213            Amount::from_sat(0)
214        } else {
215            MIN_TAPROOT_AMOUNT
216        }
217    }
218
219    pub fn anchor_amount(&self) -> Amount {
220        if self.bridge_nonstandard {
221            Amount::from_sat(0)
222        } else {
223            NON_EPHEMERAL_ANCHOR_AMOUNT
224        }
225    }
226
227    /// Checks if a block is finalized. In clementine and citrea, finality depth means the amount of confirmations needed for a block to be considered finalized.
228    /// The chain tip has 1 confirmation.
229    pub fn is_block_finalized(&self, block_height: u32, chain_tip_height: u32) -> bool {
230        if block_height > chain_tip_height {
231            return false;
232        }
233
234        chain_tip_height - block_height + 1 >= self.finality_depth
235    }
236
237    /// SP1 program keys of the network's guests and of Citrea's light client program.
238    pub fn programs(&self) -> Result<&'static NetworkPrograms, BridgeError> {
239        NetworkPrograms::for_network(self.network).ok_or(BridgeError::UnsupportedNetwork)
240    }
241
242    /// SP1 program key of the bridge guest, folded into the BitVM verifying key.
243    pub fn bridge_program_key(&self) -> Result<&[u8; 32], BridgeError> {
244        Ok(&self.programs()?.bridge.groth16)
245    }
246
247    /// SP1 program key of Citrea's light client program for the network.
248    pub fn light_client_program_key(&self) -> Result<ProgramKey, BridgeError> {
249        Ok(self.programs()?.light_client)
250    }
251
252    pub fn is_regtest(&self) -> bool {
253        self.network == Network::Regtest
254    }
255}

257fn convert_hex_string_to_bytes(hex: &str) -> Result<[u8; 32], BridgeError> {
258    let hex_decode = hex::decode(hex).wrap_err("Failed to decode hex string")?;
259    let hex_bytes: [u8; 32] = hex_decode
260        .as_slice()
261        .try_into()
262        .wrap_err("Hex string is not 32 bytes")?;
263    Ok(hex_bytes)
264}
