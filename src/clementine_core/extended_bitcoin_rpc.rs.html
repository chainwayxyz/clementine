80/// Returns the JSON-RPC error code when `error` is an error Bitcoin Core itself reported, and
81/// `None` for transport, authentication and deserialization failures.
82///
83/// Use this to tell "the node answered, and the answer is no" apart from "we could not ask the

91        _ => None,

95#[derive(Clone)]
96pub struct RetryConfig {
97    pub initial_delay_millis: u64,
98    pub max_delay: Duration,
99    pub max_attempts: usize,
100    pub backoff_multiplier: u64,
101    pub is_jitter: bool,
102    // Store the base iterator configuration
103    base_strategy: Arc<Take<ExponentialBackoff>>,
104}
105
106impl RetryConfig {
107    pub fn new(
108        initial_delay_millis: u64,
109        max_delay: Duration,
110        max_attempts: usize,
111        backoff_multiplier: u64,
112        is_jitter: bool,
113    ) -> Self {
114        // The crate use is confusing. ExponentialBackoff::from_millis defines the base,
115        // given the backoff_multiplier (this is supposed to be the initial delay), the
116        // starting factor becomes backoff_multiplier / initial_delay_millis.
117        let factor: u64 = initial_delay_millis / backoff_multiplier;
118
119        let max_attempts = std::cmp::min(max_attempts, MAX_RETRY_ATTEMPTS);
120
121        // Create the base strategy once
122        let base_strategy = Arc::new(
123            ExponentialBackoff::from_millis(backoff_multiplier)
124                .max_delay(max_delay)
125                .factor(factor)
126                .take(max_attempts),
127        );
128
129        Self {
130            initial_delay_millis,
131            max_delay,
132            max_attempts,
133            backoff_multiplier,
134            is_jitter,
135            base_strategy,
136        }
137    }
138
139    pub fn get_strategy(&self) -> Box<dyn Iterator<Item = Duration> + Send> {
140        // Clone the base strategy to get a fresh iterator with the same initial state
141        let base_strategy = (*self.base_strategy).clone();
142
143        if self.is_jitter {
144            Box::new(base_strategy.map(jitter))
145        } else {
146            Box::new(base_strategy)
147        }
148    }
149}

169/// JSON-RPC application codes for a node that is not ready.
170const RPC_CLIENT_NOT_CONNECTED: i32 = -9;
171const RPC_CLIENT_IN_INITIAL_DOWNLOAD: i32 = -10;
172const RPC_IN_WARMUP_CODE: i32 = -28;
173
174/// Returns true when an RPC application error is safe to retry.
175///
176/// Bitcoin Core emits `RPC_IN_WARMUP` from the RPC dispatcher before it calls
177/// the command. The other transient codes are command-level errors and do not
178/// prove that an arbitrary command had no effect. They are retried only when
179/// repeating the command after an unknown outcome is safe.
180fn is_retryable_rpc_code(code: i32, safe_after_unknown_outcome: bool) -> bool {
181    code == RPC_IN_WARMUP_CODE
182        || (safe_after_unknown_outcome
183            && matches!(
184                code,
185                RPC_CLIENT_NOT_CONNECTED | RPC_CLIENT_IN_INITIAL_DOWNLOAD
186            ))
187}

189/// Connection drops that can appear as I/O sources under a decode-kind reqwest
190/// error when the peer resets or aborts while the response body is still being
191/// read. Clean EOF (`UnexpectedEof`) is included; HTTP 401/404 empty bodies do
192/// not produce these kinds.
193fn is_connection_drop_io(error: &std::io::Error) -> bool {
194    use std::io::ErrorKind;
195
196    matches!(
197        error.kind(),
198        ErrorKind::ConnectionReset
199            | ErrorKind::ConnectionAborted
200            | ErrorKind::BrokenPipe
201            | ErrorKind::TimedOut
202            | ErrorKind::UnexpectedEof
203    )
204}
205
206/// Returns true if a decode-kind reqwest error was caused by the connection
207/// dropping mid-response (transport EOF) instead of a malformed or empty
208/// (e.g. HTTP 401/404) response body. reqwest's `Decoder` reports a body
209/// closed before its Content-Length completes as `is_decode`, with a
210/// `hyper::Error` (incomplete message) or a connection-drop `io::Error`
211/// source; genuine JSON syntax errors have a plain `serde_json::Error`
212/// source instead.
213fn is_transport_eof_decode(error: &reqwest::Error) -> bool {
214    if !error.is_decode() {
215        return false;
216    }
217
218    let mut source = std::error::Error::source(error);
219    while let Some(cause) = source {
220        if cause
221            .downcast_ref::<hyper::Error>()
222            .is_some_and(hyper::Error::is_incomplete_message)
223        {
224            return true;
225        }
226        if cause
227            .downcast_ref::<std::io::Error>()
228            .is_some_and(is_connection_drop_io)
229        {
230            return true;
231        }
232        source = cause.source();
233    }
234    false
235}

237/// Returns true for reqwest transport errors that are safe to retry.
238///
239/// - Connect failures never reached the server: always safe.
240/// - Mid-flight failures (request timeout, connection dropped while sending
241///   the request or while streaming the response) are only safe when
242///   `allow_mid_flight` is set, i.e. the command is idempotent: the server
243///   may have already applied a mutating command, and re-issuing it would
244///   apply it twice (double spend, extra blocks).
245/// - Unparsable responses ([`reqwest::Error::is_decode`]) are never
246///   retryable: bitcoind answers HTTP rejections such as 401 (bad auth) or
247///   404 (unknown wallet) with a non-JSON body, which surfaces as a decode
248///   error. Decode errors caused by transport EOF ([`is_transport_eof_decode`])
249///   are the exception: they are mid-flight failures.
250fn is_retryable_transport_error(
251    error: &(dyn std::error::Error + 'static),
252    allow_mid_flight: bool,
253) -> bool {
254    let Some(reqwest_error) = error.downcast_ref::<reqwest::Error>() else {
255        return false;
256    };
257
258    reqwest_error.is_connect()
259        || (allow_mid_flight
260            && (reqwest_error.is_timeout()
261                || reqwest_error.is_request()
262                || reqwest_error.is_body()
263                || is_transport_eof_decode(reqwest_error)))
264}

266/// Classifies local I/O failures without retrying known permanent errors.
267fn is_retryable_io_error(error: &std::io::Error) -> bool {
268    use std::io::ErrorKind;
269
270    match error.kind() {
271        ErrorKind::ConnectionRefused
272        | ErrorKind::ConnectionReset
273        | ErrorKind::ConnectionAborted
274        | ErrorKind::NotConnected
275        | ErrorKind::BrokenPipe
276        | ErrorKind::TimedOut
277        | ErrorKind::Interrupted
278        | ErrorKind::UnexpectedEof => true,
279        ErrorKind::PermissionDenied
280        | ErrorKind::NotFound
281        | ErrorKind::InvalidInput
282        | ErrorKind::InvalidData => false,
283        _ => true,
284    }
285}

287/// Returns true when repeating an RPC after an unknown outcome is safe and useful.
288///
289/// Commands with side effects or duplicate-in-progress failures are deliberately
290/// absent. A retry-safe command missing from this list merely means its mid-flight
291/// failures are not retried (conservative).
292fn is_safe_to_repeat_after_unknown_outcome(cmd: &str) -> bool {
293    matches!(
294        cmd,
295        // Chain and block queries
296        "getbestblockhash"
297            | "getblock"
298            | "getblockchaininfo"
299            | "getblockcount"
300            | "getblockhash"
301            | "getblockheader"
302            | "getchaintips"
303            | "getdeploymentinfo"
304            | "getindexinfo"
305            // Mempool and transaction queries
306            | "getmempoolentry"
307            | "getmempoolinfo"
308            | "getrawmempool"
309            | "getrawtransaction"
310            | "gettxout"
311            | "gettxspendingprevout"
312            | "testmempoolaccept"
313            // Fee estimation
314            | "estimatesmartfee"
315            | "estimaterawfee"
316            // Node status
317            | "getaddednodeinfo"
318            | "getmemoryinfo"
319            | "getmininginfo"
320            | "getnetworkinfo"
321            | "getpeerinfo"
322            | "getrpcinfo"
323            | "ping"
324            | "uptime"
325            // Wallet read-only queries
326            | "deriveaddresses"
327            | "getaddressinfo"
328            | "getbalance"
329            | "getbalances"
330            | "getdescriptorinfo"
331            | "gettransaction"
332            | "getwalletinfo"
333            | "listdescriptors"
334            | "listunspent"
335            | "listwalletdir"
336            | "listwallets"
337            // Stateless signing: no wallet or chain state change; a repeat
338            // after an unknown outcome returns an equally valid result.
339            | "signrawtransactionwithwallet"
340            | "walletprocesspsbt"
341            // Same hex is idempotent. Retrying avoids CPFP/withdraw funding a
342            // new tx when the first send was accepted but the response dropped.
343            | "sendrawtransaction"
344            | "submitpackage"
345    )
346}

348/// Command-aware retry classification for individual RPC calls.
349///
350/// Definite reqwest connection failures never reached the server and are safe
351/// for every command. All errors with an unknown outcome are retried only for
352/// commands that are safe to repeat. This fail-closed rule prevents retries
353/// from repeating wallet-funded sends, fee bumps, or other monetary effects.
354fn is_retryable_for_command(cmd: &str, error: &bitcoincore_rpc::Error) -> bool {
355    let safe_after_unknown_outcome = is_safe_to_repeat_after_unknown_outcome(cmd);
356
357    match error {
358        bitcoincore_rpc::Error::JsonRpc(bitcoincore_rpc::RpcError::Transport(transport_error)) => {
359            is_retryable_transport_error(transport_error.as_ref(), safe_after_unknown_outcome)
360        }
361        bitcoincore_rpc::Error::JsonRpc(bitcoincore_rpc::RpcError::Rpc(rpc_error)) => {
362            is_retryable_rpc_code(rpc_error.code, safe_after_unknown_outcome)
363        }
364        // This error does not record whether the I/O failure happened before
365        // or after dispatch, so treat its outcome as unknown.
366        bitcoincore_rpc::Error::Io(io_error) => {
367            safe_after_unknown_outcome && is_retryable_io_error(io_error)
368        }
369        _ => false,
370    }
371}

410    /// Connects to Bitcoin RPC server and checks that the node answers.
411    ///
412    /// `Client::with_timeouts` is lazy. The live check is `ping`, which goes through
413    /// [`RpcApi::call`] and the same command-aware retry path as every other RPC.
414    ///
415    /// # Parameters
416    ///
417    /// * `url` - The RPC server URL
418    /// * `user` - Username for RPC authentication
419    /// * `password` - Password for RPC authentication
420    /// * `retry_config` - Optional retry configuration. If None, uses default config.
421    ///
422    /// # Returns
423    ///
424    /// - [`Result<ExtendedBitcoinRpc>`]: A new ExtendedBitcoinRpc instance on success
425    ///
426    /// # Errors
427    ///
428    /// - [`BitcoinRPCError`]: If the client cannot be created, or ping fails after retries
429    pub async fn connect(
430        url: String,
431        user: SecretString,
432        password: SecretString,
433        retry_config: Option<RetryConfig>,
434    ) -> Result<Self> {
435        let retry_config = retry_config.unwrap_or_default();
436        let auth = Auth::UserPass(
437            user.expose_secret().to_string(),
438            password.expose_secret().to_string(),
439        );
440
441        tracing::debug!(
442            "Attempting to connect to Bitcoin RPC at {} with retry config: {:?}",
443            &url,
444            &retry_config
445        );
446        // with_timeouts only stores URL, auth, and timeouts; it does not open a socket.
447        let client = Client::with_timeouts(&url, auth, Some(RPC_REQUEST_TIMEOUT), None)
448            .await
449            .wrap_err("Failed to connect to Bitcoin RPC")?;
450
451        let rpc = Self {
452            url,
453            client: Arc::new(client),
454            retry_config,
455            #[cfg(test)]
456            cached_mining_address: Arc::new(tokio::sync::RwLock::new(None)),
457        };
458
459        tracing::debug!(
460            "Pinging Bitcoin RPC at {} to make sure it's alive",
461            &rpc.url
462        );
463        rpc.ping().await.wrap_err("Failed to ping Bitcoin RPC")?;
464
465        tracing::debug!("Connected to Bitcoin RPC successfully");
466        Ok(rpc)
467    }

516    /// Checks if an operator is still safely usable in the protocol.
517    ///
518    /// This validates:
519    /// 1. The collateral UTXO exists with the correct amount and script
520    /// 2. The round chain is followed to the current collateral position
521    /// 3. The current collateral is not spent outside the protocol
522    /// 4. Every already-passed round (ready-to-reimburse is on chain) has no leftover
523    ///    kickoff UTXOs. UnspentKickoff cannot slash those once the next round spends
524    ///    the ready-to-reimburse burn connector, so leftover kickoffs can later be used
525    ///    with already-mined reimburse connectors.
526    /// 5. If a passed-round kickoff UTXO was spent as a kickoff, that kickoff's finalizer
527    ///    must also be spent. An in-flight kickoff in a passed round is not watched by the
528    ///    round state machine. This uses the actual spending transaction of each leftover
529    ///    kickoff UTXO, including kickoffs that commit to a deposit that is not yet signed.
530    ///
531    /// `db` is the bitcoin syncer. Kickoff UTXOs of finished rounds are looked
532    /// up there; we only load block bodies for spends that might be kickoffs.
533    ///
534    /// Spent-or-not comes from the syncer, not from bitcoind. If the syncer has
535    /// no spend for a kickoff UTXO in a finalized block, treat it as still live
536    /// and reject. A short mismatch with bitcoind is fine on mainnet.
537    ///
538    /// # Returns
539    ///
540    /// - [`bool`]: `true` if the operator is still in protocol and safe to include in deposits
541    pub async fn collateral_check(
542        &self,
543        operator_data: &OperatorData,
544        kickoff_wpks: &KickoffWinternitzKeys,
545        paramset: &'static ProtocolParamset,
546        db: &Database,
547    ) -> std::result::Result<bool, BridgeError> {
548        // first check if the collateral utxo is on chain or mempool
549        let tx = self
550            .get_tx_of_txid(&operator_data.collateral_funding_outpoint.txid)
551            .await
552            .wrap_err(format!(
553                "Failed to find collateral utxo in chain for outpoint {:?}",
554                operator_data.collateral_funding_outpoint
555            ))?;
556        let collateral_outpoint = match tx
557            .output
558            .get(operator_data.collateral_funding_outpoint.vout as usize)
559        {
560            Some(output) => output,
561            None => {
562                tracing::warn!(
563                    "No output at index {} for txid {} while checking for collateral existence",
564                    operator_data.collateral_funding_outpoint.vout,
565                    operator_data.collateral_funding_outpoint.txid
566                );
567                return Ok(false);
568            }
569        };
570
571        if collateral_outpoint.value != paramset.collateral_funding_amount {
572            tracing::error!(
573                "Collateral amount for collateral {:?} is not correct: expected {}, got {}",
574                operator_data.collateral_funding_outpoint,
575                paramset.collateral_funding_amount,
576                collateral_outpoint.value
577            );
578            return Ok(false);
579        }
580
581        let operator_tpr_address =
582            create_taproot_address(&[], Some(operator_data.xonly_pk), paramset.network).0;
583
584        if collateral_outpoint.script_pubkey != operator_tpr_address.script_pubkey() {
585            tracing::error!(
586                "Collateral script pubkey for collateral {:?} is not correct: expected {}, got {}",
587                operator_data.collateral_funding_outpoint,
588                operator_tpr_address.script_pubkey(),
589                collateral_outpoint.script_pubkey
590            );
591            return Ok(false);
592        }
593
594        // we additionally check if collateral utxo is on chain (so not in mempool)
595        // on mainnet we fail if collateral utxo is not on chain because if it is in mempool,
596        // the txid of the utxo can change if the fee is bumped
597        // on other networks, we allow collateral to be in mempool to not wait for collateral to be on chain to do deposits for faster testing
598        let is_on_chain = self
599            .is_tx_on_chain(&operator_data.collateral_funding_outpoint.txid)
600            .await?;
601        if !is_on_chain {
602            return match paramset.network {
603                bitcoin::Network::Bitcoin => Ok(false),
604                _ => Ok(true),
605            };
606        }
607
608        let mut current_collateral_outpoint: OutPoint = operator_data.collateral_funding_outpoint;
609        let mut prev_ready_to_reimburse: Option<TxHandler> = None;
610        // iterate over all rounds
611        for round_idx in RoundIndex::iter_rounds(paramset.num_round_txs) {
612            // create round and ready to reimburse txs for the round
613            let txhandlers = create_round_txhandlers(
614                paramset,
615                round_idx,
616                operator_data,
617                kickoff_wpks,
618                prev_ready_to_reimburse.as_ref(),
619            )?;
620
621            let mut round_txhandler_opt = None;
622            let mut ready_to_reimburse_txhandler_opt = None;
623            for txhandler in &txhandlers {
624                match txhandler.get_transaction_type() {
625                    TransactionType::Round => round_txhandler_opt = Some(txhandler),
626                    TransactionType::ReadyToReimburse => {
627                        ready_to_reimburse_txhandler_opt = Some(txhandler)
628                    }
629                    _ => {}
630                }
631            }
632            if round_txhandler_opt.is_none() || ready_to_reimburse_txhandler_opt.is_none() {
633                return Err(eyre!(
634                    "Failed to create round and ready to reimburse txs for round {:?} for operator {}",
635                    round_idx,
636                    operator_data.xonly_pk
637                ).into());
638            }
639
640            let round_txid = round_txhandler_opt
641                .expect("Round txhandler should exist, checked above")
642                .get_cached_tx()
643                .compute_txid();
644            let is_round_tx_on_chain = self.is_tx_on_chain(&round_txid).await?;
645            if !is_round_tx_on_chain {
646                break;
647            }
648            let block_hash = self.get_blockhash_of_tx(&round_txid).await?;
649            let block_height = self
650                .get_block_info(&block_hash)
651                .await
652                .wrap_err(format!(
653                    "Failed to get block info for block hash {block_hash}"
654                ))?
655                .height;
656            if block_height < paramset.start_height as usize {
657                tracing::warn!(
658                    "Collateral utxo of operator {operator_data:?} is spent in a block before paramset start height: {block_height} < {0}",
659                    paramset.start_height
660                );
661                return Ok(false);
662            }
663            current_collateral_outpoint = OutPoint {
664                txid: round_txid,
665                vout: UtxoVout::CollateralInRound.get_vout(),
666            };
667            if round_idx == RoundIndex::Round(paramset.num_round_txs - 1) {
668                // for the last round, only check round tx, as if the operator sent the ready to reimburse tx of last round,
669                // it cannot create more kickoffs anymore
670                break;
671            }
672            let ready_to_reimburse_txhandler = ready_to_reimburse_txhandler_opt
673                .expect("Ready to reimburse txhandler should exist");
674            let ready_to_reimburse_txid =
675                ready_to_reimburse_txhandler.get_cached_tx().compute_txid();
676            let is_ready_to_reimburse_tx_on_chain =
677                self.is_tx_on_chain(&ready_to_reimburse_txid).await?;
678            if !is_ready_to_reimburse_tx_on_chain {
679                break;
680            }
681
682            // Ready-to-reimburse is on chain, so this round is passed. Leftover
683            // kickoffs can no longer be slashed via UnspentKickoff once the next
684            // round spends that burn connector.
685            let round_txhandler =
686                round_txhandler_opt.expect("Round txhandler should exist, checked above");
687            if !self
688                .passed_round_kickoffs_are_safe(
689                    round_txhandler,
690                    round_idx,
691                    paramset,
692                    operator_data,
693                    db,
694                )
695                .await?
696            {
697                return Ok(false);
698            }
699
700            current_collateral_outpoint = OutPoint {
701                txid: ready_to_reimburse_txid,
702                vout: UtxoVout::CollateralInReadyToReimburse.get_vout(),
703            };
704
705            prev_ready_to_reimburse = Some(ready_to_reimburse_txhandler.clone());
706        }
707
708        // if the collateral utxo we found latest in the round tx chain is spent, operators collateral is spent from Clementine
709        // bridge protocol, thus it is unusable and operator cannot fulfill withdrawals anymore
710        // if not spent, it should exist in chain, which is checked below
711        Ok(!self.is_utxo_spent(&current_collateral_outpoint).await?)
712    }

714    /// Returns false if a passed round still has a live kickoff UTXO, or an
715    /// in-flight kickoff whose finalizer is unspent.
716    ///
717    /// Kickoff UTXOs of this finished round must all be spent. Spent-or-not
718    /// comes from the syncer, not from bitcoind. A spend counts only if its
719    /// block is finalized. No such spend means the UTXO is still live: reject.
720    /// Do not add a `gettxout` check. A short mismatch with bitcoind is fine on
721    /// mainnet.
722    ///
723    /// A tx that spends more than one of these UTXOs cannot be a protocol
724    /// kickoff (those have a single input), so it is a burn and its block is
725    /// not loaded. Other spends are loaded from their blocks. Sequence is the
726    /// first filter: burn / UnspentKickoff set CSV, so a relative locktime
727    /// cannot be a kickoff. After that it is a protocol kickoff only if it has
728    /// exactly one input, version 3, the kickoff output layout, and the kickoff
729    /// OP_RETURN (`move_txid || this operator's xonly pk`). Any other Winternitz
730    /// spend still uses up the UTXO.
731    async fn passed_round_kickoffs_are_safe(
732        &self,
733        round_txhandler: &TxHandler,
734        round_idx: RoundIndex,
735        paramset: &'static ProtocolParamset,
736        operator_data: &OperatorData,
737        db: &Database,
738    ) -> std::result::Result<bool, BridgeError> {
739        let mut kickoff_outpoints = Vec::with_capacity(paramset.num_kickoffs_per_round);
740        for kickoff_idx in 0..paramset.num_kickoffs_per_round {
741            kickoff_outpoints.push(
742                *round_txhandler
743                    .get_spendable_output(UtxoVout::Kickoff(kickoff_idx))?
744                    .get_prev_outpoint(),
745            );
746        }
747        let mut dbtx = db.begin_transaction().await?;
748        let tip_height = db.get_max_height(Some(&mut dbtx)).await?.unwrap_or(0);
749        let mut spends = db
750            .get_canonical_spending_txids(Some(&mut dbtx), &kickoff_outpoints)
751            .await?;
752        // A reorg of a non-final spend revives the UTXO after deposits are signed,
753        // when the burn connectors are already spent by the next round.
754        spends.retain(|_, (_, _, height)| paramset.is_block_finalized(*height, tip_height));
755
756        let mut spend_count: HashMap<Txid, usize> = HashMap::new();
757        for (spending_txid, _, _) in spends.values() {
758            *spend_count.entry(*spending_txid).or_default() += 1;
759        }
760
761        let mut to_classify = Vec::new();
762        let mut wanted_txids = HashSet::new();
763        let mut unique_blockhashes = HashSet::new();
764        for (kickoff_idx, kickoff_outpoint) in kickoff_outpoints.iter().enumerate() {
765            let Some((spending_txid, blockhash, _)) = spends.get(kickoff_outpoint) else {
766                tracing::error!(
767                    ?operator_data,
768                    ?round_idx,
769                    kickoff_idx,
770                    %kickoff_outpoint,
771                    "Passed round has a kickoff UTXO without a finalized spend; operator is not safe to include in deposits"
772                );
773                dbtx.commit().await?;
774                return Ok(false);
775            };
776            // A tx that consumes more than one leftover kickoff in this round cannot
777            // be a protocol kickoff (those have a single input).
778            if spend_count[spending_txid] > 1 {
779                continue;
780            }
781            to_classify.push((*kickoff_outpoint, *spending_txid));
782            wanted_txids.insert(*spending_txid);
783            unique_blockhashes.insert(*blockhash);
784        }
785
786        // It is assumed that the number of unique block hashes is small, so we can fetch all the transactions in one query.
787        // It will be obvious on mainnet if an operator intentionally tries to grief this.
788        let spending_txs = db
789            .get_txs_from_block_hashes(Some(&mut dbtx), unique_blockhashes, &wanted_txids)
790            .await?;
791
792        let mut finalizer_outpoints = Vec::new();
793        for (kickoff_outpoint, spending_txid) in to_classify {
794            let Some(spending_tx) = spending_txs.get(&spending_txid) else {
795                return Err(eyre!(
796                    "Canonical spending txid {spending_txid} was indexed but missing from its block"
797                )
798                .into());
799            };
800            if spending_is_protocol_kickoff(kickoff_outpoint, spending_tx, operator_data.xonly_pk)?
801            {
802                finalizer_outpoints.push(OutPoint {
803                    txid: spending_txid,
804                    vout: UtxoVout::KickoffFinalizer.get_vout(),
805                });
806            }
807        }
808
809        if !finalizer_outpoints.is_empty() {
810            let spent_finalizers = db
811                .get_canonical_spending_txids(Some(&mut dbtx), &finalizer_outpoints)
812                .await?;
813            if let Some(finalizer) = finalizer_outpoints.iter().find(|outpoint| {
814                !spent_finalizers
815                    .get(outpoint)
816                    .is_some_and(|(_, _, height)| paramset.is_block_finalized(*height, tip_height))
817            }) {
818                tracing::error!(
819                    ?operator_data,
820                    ?round_idx,
821                    %finalizer,
822                    "Passed round kickoff UTXO was spent as a kickoff but the kickoff finalizer has no finalized spend"
823                );
824                dbtx.commit().await?;
825                return Ok(false);
826            }
827        }
828        dbtx.commit().await?;
829        Ok(true)
830    }

1035    /// A helper fn to safely mine blocks while waiting for all actors to be synced
1036    /// If CitreaE2EData is provided and there are multiple DA nodes, it will additionally perform a reorg. Reorg will cause chain size to increase by number of reorged blocks in addition to block_num.
1037    #[cfg(test)]
1038    pub async fn mine_blocks_while_synced<C: CitreaClientT>(
1039        &self,
1040        block_num: u64,
1041        actors: &TestActors<C>,
1042        e2e: Option<&CitreaE2EData<'_>>,
1043    ) -> Result<Vec<BlockHash>> {
1044        match e2e {
1045            // do a reorg if there are multiple nodes, the actors are currently always using node 0
1046            Some(e2e) if e2e.bitcoin_nodes.iter().count() > 1 => {
1047                use bitcoin::secp256k1::rand::{thread_rng, Rng};
1048                e2e.bitcoin_nodes
1049                    .disconnect_nodes()
1050                    .await
1051                    .map_err(|e| eyre::eyre!("Failed to disconnect nodes: {}", e))?;
1052                let reorg_blocks =
1053                    thread_rng().gen_range(0..e2e.config.protocol_paramset().finality_depth as u64);
1054                let da0 = e2e.bitcoin_nodes.get(0).expect("node 0 should exist");
1055                let da1 = e2e.bitcoin_nodes.get(1).expect("node 1 should exist");
1056
1057                let mut mined_blocks = Vec::new();
1058                while mined_blocks.len() < reorg_blocks as usize {
1059                    if !are_all_nodes_synced(self, actors).await? {
1060                        // wait until they are synced
1061                        tokio::time::sleep(std::time::Duration::from_millis(300)).await;
1062                        continue;
1063                    }
1064                    da0.generate(1)
1065                        .await
1066                        .wrap_err("Failed to generate blocks")?;
1067                    // da1 will be canonical
1068                    let new_blocks = da1
1069                        .generate(1)
1070                        .await
1071                        .wrap_err("Failed to generate blocks")?;
1072                    mined_blocks.extend(new_blocks);
1073                }
1074                mined_blocks.extend(
1075                    da1.generate(1)
1076                        .await
1077                        .wrap_err("Failed to generate blocks")?,
1078                );
1079                // connect and reorg, da0's blocks will be non-canonical
1080                e2e.bitcoin_nodes
1081                    .connect_nodes()
1082                    .await
1083                    .map_err(|e| eyre::eyre!("Failed to connect nodes: {}", e))?;
1084                e2e.bitcoin_nodes
1085                    .wait_for_sync(None)
1086                    .await
1087                    .map_err(|e| eyre::eyre!("Failed to wait for sync: {}", e))?;
1088                // mined blocks has to be block_num higher than reorg_blocks + 1, because this callers of this fn expects
1089                // chain size to increase by at least block_num starting from where cpfp fee payer is in mempool.
1090                while mined_blocks.len() != (reorg_blocks + block_num + 1) as usize {
1091                    if !are_all_nodes_synced(self, actors).await? {
1092                        // wait until they are synced
1093                        tokio::time::sleep(std::time::Duration::from_millis(300)).await;
1094                        continue;
1095                    }
1096                    mined_blocks.extend(self.mine_blocks(1).await?);
1097                }
1098                Ok(mined_blocks)
1099            }
1100            _ => {
1101                // do not do a reorg here, just mine blocks
1102                let mut mined_blocks = Vec::new();
1103                while mined_blocks.len() < block_num as usize {
1104                    if !are_all_nodes_synced(self, actors).await? {
1105                        // wait until they are synced
1106                        tokio::time::sleep(std::time::Duration::from_millis(300)).await;
1107                        continue;
1108                    }
1109                    let new_blocks = self.mine_blocks(1).await?;
1110                    mined_blocks.extend(new_blocks);
1111                }
1112                Ok(mined_blocks)
1113            }
1114        }
1115    }

1243    /// Builds and signs a Replace-By-Fee (RBF) replacement of an unconfirmed
1244    /// transaction that meets or exceeds a target fee rate. It does not
1245    /// broadcast the replacement, so the caller can persist it first.
1246    ///
1247    /// It works as follows:
1248    /// 1. If the transaction is already confirmed, returns Err(TransactionAlreadyInBlock)
1249    /// 2. If the current fee rate is already >= the requested fee rate, returns `None`
1250    /// 3. Otherwise, increases the fee rate by adding the node's incremental fee to the current fee rate,
1251    ///    builds the replacement with `psbtbumpfee`, then signs and finalizes it with the wallet
1252    ///
1253    /// Note: This function currently only supports fee payer TXs.
1254    ///
1255    /// # Arguments
1256    /// * `txid` - The transaction ID to bump
1257    /// * `fee_rate` - The target fee rate to achieve
1258    ///
1259    /// # Returns
1260    ///
1261    /// - [`Option<Transaction>`]: The signed replacement, or `None` if no bump was needed.
1262    ///
1263    /// # Errors
1264    ///
1265    ///  * `TransactionAlreadyInBlock` - If the transaction is already confirmed
1266    /// * `BumpFeeUTXOSpent` - If the UTXO being spent by the transaction is already spent
1267    /// * `BumpFeeError` - For other errors with fee bumping
1268    pub async fn create_fee_bump_tx(
1269        &self,
1270        txid: Txid,
1271        fee_rate: FeeRate,
1272    ) -> Result<Option<Transaction>> {
1273        // Check if transaction is already confirmed
1274        let transaction_info = self
1275            .get_transaction(&txid, None)
1276            .await
1277            .wrap_err("Failed to get transaction")?;
1278        if let Some(blockhash) = transaction_info.info.blockhash {
1279            return Err(BitcoinRPCError::TransactionAlreadyInBlock(blockhash));
1280        }
1281
1282        // Calculate current fee rate
1283        let tx = transaction_info
1284            .transaction()
1285            .wrap_err("Failed to get transaction")?;
1286        let tx_weight = tx.weight().to_wu();
1287        let current_fee_sat = u64::try_from(
1288            transaction_info
1289                .fee
1290                .expect("Fee should be present")
1291                .to_sat()
1292                .abs(),
1293        )
1294        .wrap_err("Failed to convert fee to sat")?;
1295
1296        let current_fee_rate_sat_kwu = current_fee_sat as f64 * 1000.0 / tx_weight as f64;
1297
1298        tracing::trace!(
1299            "Bump fee with fee rate txid: {txid} - Current fee sat: {current_fee_sat} - current fee rate: {current_fee_rate_sat_kwu}"
1300        );
1301
1302        // If current fee rate is already sufficient, no replacement is needed
1303        if current_fee_rate_sat_kwu >= fee_rate.to_sat_per_kwu() as f64 {
1304            return Ok(None);
1305        }
1306
1307        tracing::trace!(
1308            "Bump fee with fee rate txid: {txid} - Current fee rate: {current_fee_rate_sat_kwu} sat/kwu, target fee rate: {fee_rate} sat/kwu"
1309        );
1310
1311        // Get node's incremental fee to determine how much to increase
1312        let network_info = self
1313            .get_network_info()
1314            .await
1315            .wrap_err("Failed to get network info")?;
1316        // incremental fee is in BTC/kvB
1317        let incremental_fee = network_info.incremental_fee;
1318        // Convert from sat/kvB to sat/kwu by dividing by 4.0, since 1 kvB = 4 kwu.
1319        let incremental_fee_rate_sat_kwu = incremental_fee.to_sat() as f64 / 4.0;
1320
1321        // Calculate new fee rate by adding incremental fee to current fee rate, or use the target fee rate if it's higher
1322        let new_fee_rate = FeeRate::from_sat_per_kwu(std::cmp::max(
1323            (current_fee_rate_sat_kwu + incremental_fee_rate_sat_kwu).ceil() as u64,
1324            fee_rate.to_sat_per_kwu(),
1325        ));
1326
1327        tracing::debug!(
1328            "Bumping fee for txid: {txid} from {current_fee_rate_sat_kwu} to {new_fee_rate} with incremental fee {incremental_fee_rate_sat_kwu} - Final fee rate: {new_fee_rate}, current chain fee rate: {fee_rate}"
1329        );
1330
1331        // Call Bitcoin Core's psbtbumpfee RPC
1332        let bump_fee_result = match self
1333            .psbt_bump_fee(
1334                &txid,
1335                Some(&bitcoincore_rpc::json::BumpFeeOptions {
1336                    fee_rate: Some(bitcoincore_rpc::json::FeeRate::per_vbyte(Amount::from_sat(
1337                        new_fee_rate.to_sat_per_vb_ceil(),
1338                    ))),
1339                    replaceable: Some(true),
1340                    ..Default::default()
1341                }),
1342            )
1343            .await
1344        {
1345            Ok(bump_fee_result) => bump_fee_result,
1346            // Attempt to parse the error message to get the outpoint if the UTXO is already spent
1347            Err(e) => match e {
1348                bitcoincore_rpc::Error::JsonRpc(json_rpc_error) => match json_rpc_error {
1349                    bitcoincore_rpc::RpcError::Rpc(rpc_error) => {
1350                        if let Some((outpoint_str, _)) =
1351                            rpc_error.message.split_once(" is already spent")
1352                        {
1353                            let outpoint = OutPoint::from_str(outpoint_str)
1354                                .wrap_err(BitcoinRPCError::BumpFeeError(txid, fee_rate))?;
1355
1356                            return Err(BitcoinRPCError::BumpFeeUTXOSpent(outpoint));
1357                        }
1358
1359                        return Err(eyre::eyre!("{:?}", rpc_error)
1360                            .wrap_err(BitcoinRPCError::BumpFeeError(txid, fee_rate))
1361                            .into());
1362                    }
1363                    _ => {
1364                        return Err(eyre::eyre!(json_rpc_error)
1365                            .wrap_err(BitcoinRPCError::BumpFeeError(txid, fee_rate))
1366                            .into());
1367                    }
1368                },
1369                _ => {
1370                    return Err(eyre::eyre!(e)
1371                        .wrap_err(BitcoinRPCError::BumpFeeError(txid, fee_rate))
1372                        .into());
1373                }
1374            },
1375        };
1376
1377        let psbt = bump_fee_result
1378            .psbt
1379            .ok_or_eyre("Failed to get PSBT from bump_fee_result")?;
1380        let processed = self
1381            .wallet_process_psbt(&psbt, Some(true), None, None)
1382            .await
1383            .wrap_err("Failed to sign bumped tx with wallet")?;
1384        if !processed.complete {
1385            return Err(eyre!("Wallet could not fully sign the replacement of {txid}").into());
1386        }
1387        let finalized = self
1388            .finalize_psbt(&processed.psbt, None)
1389            .await
1390            .wrap_err("Failed to finalize bumped tx")?;
1391        if !finalized.complete {
1392            return Err(eyre!("Could not finalize the replacement of {txid}").into());
1393        }
1394        let bumped_tx: Transaction = bitcoin::consensus::deserialize(
1395            &finalized
1396                .hex
1397                .ok_or_eyre("finalizepsbt returned no transaction")?,
1398        )
1399        .wrap_err("Failed to deserialize bumped tx")?;
1400
1401        Ok(Some(bumped_tx))
1402    }

1443    /// Gets the current recommended fee rate in sat/vb from Mempool Space and Bitcoin Core and selects the minimum.
1444    /// For Regtest and Signet, it uses a fixed fee rate of 1 sat/vB.
1445    /// # Logic
1446    /// *   **Regtest:** Uses a fixed fee rate of 1 sat/vB for simplicity.
1447    /// *   **Mainnet, Testnet4 and Signet:** Fetches fee rates from both Mempool Space API and Bitcoin Core RPC and takes the minimum.
1448    /// *   **Hard Cap:** Applies a hard cap from configuration to prevent excessive fees.
1449    /// # Fallbacks
1450    /// *   If one source fails, it uses the other.
1451    /// *   If both fail, it falls back to a default of 1 sat/vB.
1452    pub async fn get_fee_rate(
1453        &self,
1454        network: Network,
1455        mempool_api_host: &Option<String>,
1456        mempool_api_endpoint: &Option<String>,
1457        mempool_fee_rate_multiplier: u64,
1458        mempool_fee_rate_offset_sat_kvb: u64,
1459        fee_rate_hard_cap: u64,
1460    ) -> Result<FeeRate> {
1461        match network {
1462            // Regtest use a fixed, low fee rate.
1463            Network::Regtest => {
1464                tracing::debug!("Using fixed fee rate of 1 sat/vB for {network} network");
1465                Ok(FeeRate::from_sat_per_vb_u32(1))
1466            }
1467
1468            // Mainnet and Testnet4 fetch fees from Mempool Space and Bitcoin Core RPC.
1469            Network::Bitcoin | Network::Testnet4 | Network::Signet => {
1470                tracing::debug!("Fetching fee rate for {network} network...");
1471
1472                // Fetch fees from both mempool.space and Bitcoin Core RPC
1473                let mempool_fee = get_fee_rate_from_mempool_space(
1474                    mempool_api_host,
1475                    mempool_api_endpoint,
1476                    network,
1477                )
1478                .await;
1479
1480                let rpc_fee = timeout(Duration::from_secs(30), self.estimate_smart_fee(1, None))
1481                    .await
1482                    .map_err(|_| eyre!("RPC estimate_smart_fee timed out after 30 seconds"))
1483                    .and_then(|result| {
1484                        result.wrap_err("Failed to estimate smart fee using Bitcoin Core RPC")
1485                    })
1486                    .and_then(|estimate| {
1487                        estimate.fee_rate.ok_or_else(|| {
1488                            eyre!("Failed to extract fee rate from Bitcoin Core RPC response")
1489                        })
1490                    });
1491
1492                // Use the minimum of both fee sources, with fallback logic, carefully avoiding overflow
1493                let selected_fee_amount = match (mempool_fee, rpc_fee) {
1494                    (Ok(mempool_amt), Ok(rpc_amt)) => {
1495                        // Use checked arithmetic to avoid overflow
1496                        let multiplier = mempool_fee_rate_multiplier;
1497                        let offset = mempool_fee_rate_offset_sat_kvb;
1498                        let rpc_amt_sat = rpc_amt.to_sat();
1499
1500                        let threshold_sat = multiplier
1501                            .checked_mul(rpc_amt_sat)
1502                            .and_then(|v| v.checked_add(offset))
1503                            .ok_or_else(|| {
1504                                eyre!("Overflow when calculating threshold_sat in fee selection ({multiplier} * {rpc_amt_sat} + {offset})")
1505                            })?;
1506
1507                        let threshold = Amount::from_sat(threshold_sat);
1508
1509                        if mempool_amt <= threshold {
1510                            tracing::debug!(
1511                                "Selected mempool.space fee rate: {} sat/kvB (mempool: {}, rpc: {}, threshold: {})",
1512                                mempool_amt.to_sat(),
1513                                mempool_amt.to_sat(),
1514                                rpc_amt.to_sat(),
1515                                threshold
1516                            );
1517                            mempool_amt
1518                        } else {
1519                            tracing::debug!(
1520                                "Selected Bitcoin Core RPC fee rate: {} sat/kvB (mempool: {}, rpc: {}, threshold: {})",
1521                                rpc_amt.to_sat(),
1522                                mempool_amt.to_sat(),
1523                                rpc_amt.to_sat(),
1524                                threshold
1525                            );
1526                            rpc_amt
1527                        }
1528                    }
1529                    (Ok(mempool_amt), Err(rpc_err)) => {
1530                        tracing::debug!(
1531                            "RPC fee estimation failed, using mempool.space: {:#}",
1532                            rpc_err
1533                        );
1534                        mempool_amt
1535                    }
1536                    (Err(mempool_err), Ok(rpc_amt)) => {
1537                        tracing::debug!(
1538                            "Mempool.space fee fetch failed, using Bitcoin Core RPC: {:#}",
1539                            mempool_err
1540                        );
1541                        rpc_amt
1542                    }
1543                    (Err(mempool_err), Err(rpc_err)) => {
1544                        tracing::warn!(
1545                            "Both fee sources failed (mempool: {:#}, rpc: {:#}), using default of 1 sat/vB",
1546                            mempool_err,
1547                            rpc_err
1548                        );
1549                        Amount::from_sat(1000) // 1 sat/vB * 1000 = 1000 sat/kvB
1550                    }
1551                };
1552
1553                // Convert sat/kvB to sat/vB and apply hard cap
1554                let mut fee_sat_kvb = selected_fee_amount.to_sat();
1555
1556                // Apply hard cap from config
1557                if fee_sat_kvb > fee_rate_hard_cap * 1000 {
1558                    tracing::debug!(
1559                        "Fee rate {} sat/kvb exceeds hard cap {} sat/kvb, using hard cap",
1560                        fee_sat_kvb,
1561                        fee_rate_hard_cap * 1000
1562                    );
1563                    fee_sat_kvb = fee_rate_hard_cap * 1000;
1564                }
1565
1566                tracing::debug!("Final fee rate: {} sat/kvb", fee_sat_kvb);
1567                Ok(FeeRate::from_sat_per_kwu(fee_sat_kvb.div_ceil(4)))
1568            }
1569
1570            // All other network types are unsupported.
1571            _ => Err(eyre!(
1572                "Fee rate estimation is not supported for network: {:?}",
1573                network
1574            )
1575            .into()),
1576        }
1577    }

1580/// Fetches the current recommended fee rate from the provider. Currently only supports
1581/// Mempool Space API.
1582/// This function is used to get the fee rate in sat/vkb (satoshis per kilovbyte).
1583/// See [Mempool Space API](https://mempool.space/docs/api/rest#get-recommended-fees) for more details.
1584async fn get_fee_rate_from_mempool_space(
1585    url: &Option<String>,
1586    endpoint: &Option<String>,
1587    network: Network,
1588) -> Result<Amount> {
1589    let url = url
1590        .as_ref()
1591        .ok_or_else(|| eyre!("Fee rate API host is not configured"))?;
1592
1593    let endpoint = endpoint
1594        .as_ref()
1595        .ok_or_else(|| eyre!("Fee rate API endpoint is not configured"))?;
1596
1597    let url = match network {
1598        Network::Bitcoin => format!(
1599            // If the variables are not, return Error to fallback to Bitcoin Core RPC.
1600            "{url}{endpoint}",
1601        ),
1602        Network::Testnet4 => format!("{url}testnet4/{endpoint}"),
1603        // Return early with error for unsupported networks
1604        Network::Signet => {
1605            tracing::debug!("You should use Citrea signet url for mempool.space");
1606            format!("{url}{endpoint}")
1607        }
1608        _ => return Err(eyre!("Unsupported network for mempool.space: {network:?}").into()),
1609    };
1610
1611    let retry_config = RetryConfig::new(250, Duration::from_secs(5), 4, 2, true);
1612
1613    let retry_strategy = retry_config.get_strategy();
1614
1615    // Retry predicate: only retry on timeouts, connect errors, and 5xx/429 statuses.
1616    let should_retry = |e: &FeeErr| match e {
1617        FeeErr::Timeout => true,
1618        FeeErr::Transport(req_err) => req_err.is_timeout() || req_err.is_connect(),
1619        FeeErr::Status(code) => code.is_server_error() || *code == StatusCode::TOO_MANY_REQUESTS,
1620        FeeErr::JsonDecode(_) | FeeErr::MissingField => false,
1621    };
1622
1623    let fee_sat_per_vb: u64 = RetryIf::spawn(
1624        retry_strategy,
1625        || {
1626            let url = url.clone();
1627            async move {
1628                let resp = timeout(Duration::from_secs(5), reqwest::get(&url))
1629                    .await
1630                    .map_err(|_| FeeErr::Timeout)?
1631                    .map_err(FeeErr::Transport)?;
1632
1633                let status = resp.status();
1634                if !status.is_success() {
1635                    return Err(FeeErr::Status(status));
1636                }
1637
1638                let json: serde_json::Value = timeout(Duration::from_secs(5), resp.json())
1639                    .await
1640                    .map_err(|_| FeeErr::Timeout)?
1641                    .map_err(FeeErr::JsonDecode)?;
1642
1643                json.get("fastestFee")
1644                    .and_then(|fee| fee.as_u64())
1645                    .ok_or(FeeErr::MissingField)
1646            }
1647        },
1648        should_retry,
1649    )
1650    .await
1651    .map_err(|e| eyre::eyre!(e))
1652    .wrap_err_with(|| format!("Failed to fetch/parse fees from {url}"))?;
1653
1654    // The API returns the fee rate in sat/vB. We multiply by 1000 to get sat/kvB.
1655    let fee_rate = Amount::from_sat(fee_sat_per_vb * 1000);
1656
1657    Ok(fee_rate)
1658}

1660/// Challenge, KickoffFinalizer, Reimburse, Disprove, LatestBlockhash, asserts,
1661/// OP_RETURN, and the CPFP anchor. Watchtower pairs are extra and deposit-dependent.
1662fn min_kickoff_output_count() -> usize {

1666/// True if `spending_tx` spends `kickoff_outpoint` as a protocol kickoff.
1667///
1668/// Sequence is the first filter: the 1-block CSV burn / UnspentKickoff path
1669/// cannot be a kickoff. A kickoff spends only that UTXO, so more than one
1670/// input cannot be a kickoff. After that it must be version 3, have the
1671/// protocol output count, and include the kickoff OP_RETURN ending in this
1672/// operator's xonly pk. Missing the spent input is an invariant error: the
1673/// caller only passes transactions indexed as spending this outpoint.
1674fn spending_is_protocol_kickoff(
1675    kickoff_outpoint: OutPoint,
1676    spending_tx: &Transaction,
1677    operator_xonly_pk: XOnlyPublicKey,
1678) -> std::result::Result<bool, BridgeError> {
1679    let Some(txin) = spending_tx
1680        .input
1681        .iter()
1682        .find(|txin| txin.previous_output == kickoff_outpoint)
1683    else {
1684        return Err(eyre!(
1685            "Spending transaction {} does not include kickoff outpoint {}",
1686            spending_tx.compute_txid(),
1687            kickoff_outpoint
1688        )
1689        .into());
1690    };
1691    if txin.sequence.is_relative_lock_time() {
1692        return Ok(false);
1693    }
1694    if spending_tx.input.len() != 1 {
1695        return Ok(false);
1696    }
1697    if spending_tx.version != NON_STANDARD_V3 {
1698        return Ok(false);
1699    }
1700    if spending_tx.output.len() < min_kickoff_output_count() {
1701        return Ok(false);
1702    }
1703    let operator_pk = operator_xonly_pk.serialize();
1704    Ok(spending_tx.output.iter().any(|output| {
1705        parse_op_return_data(&output.script_pubkey)
1706            .is_some_and(|data| data.len() == 64 && data[32..] == operator_pk)
1707    }))
1708}
