13const SETUP_DOMAIN: &[u8] = b"clementine/operator_setup/v1";
14const DEPOSIT_KEYS_DOMAIN: &[u8] = b"clementine/operator_deposit_keys/v1";

16struct Transcript {
17    bytes: Vec<u8>,
18}
19
20impl Transcript {
21    fn new(domain: &[u8]) -> Self {
22        let mut transcript = Self { bytes: Vec::new() };
23        transcript.push_bytes(domain);
24        transcript
25    }
26
27    fn push_bytes(&mut self, bytes: &[u8]) {
28        self.bytes
29            .extend_from_slice(&(bytes.len() as u64).to_le_bytes());
30        self.bytes.extend_from_slice(bytes);
31    }
32
33    fn push_u32(&mut self, value: u32) {
34        self.bytes.extend_from_slice(&value.to_le_bytes());
35    }
36
37    fn push_xonly_pk(&mut self, pk: XOnlyPublicKey) {
38        self.push_bytes(&pk.serialize());
39    }
40
41    fn push_outpoint(&mut self, outpoint: OutPoint) {
42        self.push_bytes(&outpoint.txid.to_byte_array());
43        self.push_u32(outpoint.vout);
44    }
45
46    fn push_address(&mut self, address: &Address) {
47        self.push_bytes(address.to_string().as_bytes());
48    }
49
50    fn push_winternitz_public_keys(&mut self, keys: &[winternitz::PublicKey]) {
51        self.bytes
52            .extend_from_slice(&(keys.len() as u64).to_le_bytes());
53        for key in keys {
54            self.bytes
55                .extend_from_slice(&(key.len() as u64).to_le_bytes());
56            for digit in key {
57                self.push_bytes(digit);
58            }
59        }
60    }
61
62    fn push_schnorr_sigs(&mut self, sigs: &[Signature]) {
63        self.bytes
64            .extend_from_slice(&(sigs.len() as u64).to_le_bytes());
65        for sig in sigs {
66            self.push_bytes(&sig.serialize());
67        }
68    }
69
70    fn push_hashes(&mut self, hashes: &[PublicHash]) {
71        self.bytes
72            .extend_from_slice(&(hashes.len() as u64).to_le_bytes());
73        for hash in hashes {
74            self.push_bytes(hash);
75        }
76    }
77
78    fn push_deposit_data(&mut self, deposit_data: &DepositData) -> Result<(), BridgeError> {
79        let mut deposit_data = deposit_data.clone();
80        // sort here so that the array ordering doesn't cause mismatches
81        deposit_data.actors.verifiers.sort();
82        deposit_data.actors.watchtowers.sort();
83        deposit_data.actors.operators.sort();
84        let deposit_params: DepositParams = deposit_data.into();
85        self.push_bytes(&deposit_params.encode_to_vec());
86        Ok(())
87    }
88
89    fn digest(self) -> [u8; 32] {
90        sha256::Hash::hash(&self.bytes).to_byte_array()
91    }
92}
93
94fn sign_digest(actor: &Actor, digest: [u8; 32]) -> Signature {
95    SECP.sign_schnorr(&Message::from_digest(digest), &actor.keypair)
96}
97
98fn verify_digest(
99    operator_xonly_pk: XOnlyPublicKey,
100    digest: [u8; 32],
101    signature: &Signature,
102) -> Result<(), BridgeError> {
103    SECP.verify_schnorr(signature, &Message::from_digest(digest), &operator_xonly_pk)
104        .map_err(|_| eyre::eyre!("Operator auth signature verification failed").into())
105}
106
107fn setup_digest(
108    operator_xonly_pk: XOnlyPublicKey,
109    collateral_funding_outpoint: OutPoint,
110    wallet_reimburse_address: &Address,
111    kickoff_winternitz_public_keys: &[winternitz::PublicKey],
112    unspent_kickoff_sigs: &[Signature],
113) -> [u8; 32] {
114    let mut transcript = Transcript::new(SETUP_DOMAIN);
115    transcript.push_xonly_pk(operator_xonly_pk);
116    transcript.push_outpoint(collateral_funding_outpoint);
117    transcript.push_address(wallet_reimburse_address);
118    transcript.push_winternitz_public_keys(kickoff_winternitz_public_keys);
119    transcript.push_schnorr_sigs(unspent_kickoff_sigs);
120    transcript.digest()
121}
122
123fn deposit_keys_digest(
124    operator_xonly_pk: XOnlyPublicKey,
125    deposit_data: &DepositData,
126    winternitz_public_keys: &[winternitz::PublicKey],
127    challenge_ack_hashes: &[PublicHash],
128) -> Result<[u8; 32], BridgeError> {
129    let mut transcript = Transcript::new(DEPOSIT_KEYS_DOMAIN);
130    transcript.push_xonly_pk(operator_xonly_pk);
131    transcript.push_deposit_data(deposit_data)?;
132    transcript.push_winternitz_public_keys(winternitz_public_keys);
133    transcript.push_hashes(challenge_ack_hashes);
134    Ok(transcript.digest())
135}

137/// Signs the operator setup data that the aggregator forwards during setup.
138pub fn sign_operator_setup(
139    actor: &Actor,
140    collateral_funding_outpoint: OutPoint,
141    wallet_reimburse_address: &Address,
142    kickoff_winternitz_public_keys: &[winternitz::PublicKey],
143    unspent_kickoff_sigs: &[Signature],
144) -> Signature {
145    sign_digest(
146        actor,
147        setup_digest(
148            actor.xonly_public_key,
149            collateral_funding_outpoint,
150            wallet_reimburse_address,
151            kickoff_winternitz_public_keys,
152            unspent_kickoff_sigs,
153        ),
154    )
155}
156
157/// Verifies the operator setup data before it is persisted by a verifier.
158pub fn verify_operator_setup(
159    operator_xonly_pk: XOnlyPublicKey,
160    collateral_funding_outpoint: OutPoint,
161    wallet_reimburse_address: &Address,
162    kickoff_winternitz_public_keys: &[winternitz::PublicKey],
163    unspent_kickoff_sigs: &[Signature],
164    signature: &Signature,
165) -> Result<(), BridgeError> {
166    verify_digest(
167        operator_xonly_pk,
168        setup_digest(
169            operator_xonly_pk,
170            collateral_funding_outpoint,
171            wallet_reimburse_address,
172            kickoff_winternitz_public_keys,
173            unspent_kickoff_sigs,
174        ),
175        signature,
176    )
177}

179/// Signs per-deposit operator keys that the aggregator forwards to verifiers.
180pub fn sign_operator_deposit_keys(
181    actor: &Actor,
182    deposit_data: &DepositData,
183    winternitz_public_keys: &[winternitz::PublicKey],
184    challenge_ack_hashes: &[PublicHash],
185) -> Result<Signature, BridgeError> {
186    Ok(sign_digest(
187        actor,
188        deposit_keys_digest(
189            actor.xonly_public_key,
190            deposit_data,
191            winternitz_public_keys,
192            challenge_ack_hashes,
193        )?,
194    ))
195}

197/// Verifies per-deposit operator keys before they are persisted by a verifier.
198pub fn verify_operator_deposit_keys(
199    operator_xonly_pk: XOnlyPublicKey,
200    deposit_data: &DepositData,
201    winternitz_public_keys: &[winternitz::PublicKey],
202    challenge_ack_hashes: &[PublicHash],
203    signature: &Signature,
204) -> Result<(), BridgeError> {
205    verify_digest(
206        operator_xonly_pk,
207        deposit_keys_digest(
208            operator_xonly_pk,
209            deposit_data,
210            winternitz_public_keys,
211            challenge_ack_hashes,
212        )?,
213        signature,
214    )
