1//! # Bitcoin Script Construction
2//!
3//! This module provides a collection of builders for creating various Bitcoin
4//! scripts utilized within the Clementine bridge. It defines a `SpendableScript`
5//! trait, implemented by specific script structures (e.g., `CheckSig`,
6//! `WinternitzCommit`, `TimelockScript`, `BaseDepositScript`) to standardize
7//! script generation and witness creation.
8//!
9//! Each script builder offers:
10//! - A constructor to initialize the script with its specific parameters.
11//! - A method to convert the script structure into a `bitcoin::ScriptBuf`.
12//! - A method to generate the corresponding `bitcoin::Witness` required to spend
13//!   an output locked with this script.
14//!
15//! The module also includes `ScriptKind`, an enum to differentiate between various
16//! spendable script types, facilitating dynamic dispatch and script management.
17//! Helper functions like `extract_winternitz_commits` are provided for parsing
18//! specific data committed using witnernitz keys from witness.

46fn parse_winternitz_digit(digit: &[u8], max_digit: u8) -> Result<u8> {
47    let parsed = bitcoin::script::read_scriptint_non_minimal(digit)
48        .wrap_err("Failed to parse Winternitz digit as Bitcoin script integer")?;
49    if parsed < 0 {
50        return Err(eyre::eyre!("Winternitz digit cannot be negative: {parsed}"));
51    }
52
53    Ok(std::cmp::min(parsed, i64::from(max_digit)) as u8)
54}

56/// Re-encodes a Winternitz digit as the minimal script number of the value that
57/// the on-chain verifier uses after `OP_MIN(digit, max_digit)`.
58fn canonicalize_winternitz_digit(digit: &[u8], max_digit: u8) -> Result<Vec<u8>> {
59    let value = parse_winternitz_digit(digit, max_digit)?;
60    let mut scriptint_buf = [0u8; 8];
61    let scriptint_len = bitcoin::script::write_scriptint(&mut scriptint_buf, i64::from(value));
62    Ok(scriptint_buf[..scriptint_len].to_vec())
63}

65/// Extracts the committed data from the witness.
66/// Note: The function is hardcoded for winternitz_log_d = 4 currently, will not work for others.
67pub fn extract_winternitz_commits(
68    witness: Witness,
69    wt_derive_paths: &[WinternitzDerivationPath],
70    paramset: &'static ProtocolParamset,
71) -> Result<Vec<Vec<u8>>> {
72    if paramset.winternitz_log_d != 4 {
73        return Err(eyre::eyre!("Only winternitz_log_d = 4 is supported"));
74    }
75    let digit_base = 1u8
76        .checked_shl(paramset.winternitz_log_d)
77        .ok_or_else(|| eyre::eyre!("Winternitz digit base overflow"))?;
78    let expected_max_digit = digit_base - 1;
79
80    let mut commits: Vec<Vec<u8>> = Vec::new();
81    let mut cur_witness_iter = witness.into_iter().skip(1);
82
83    for wt_path in wt_derive_paths.iter().rev() {
84        let wt_params = wt_path.get_params();
85        let actual_max_digit =
86            u8::try_from(wt_params.max_digit()).wrap_err("Failed to convert max digit to u8")?;
87        if actual_max_digit != expected_max_digit {
88            return Err(eyre::eyre!(
89                "Winternitz path log2_base must match paramset winternitz_log_d: max_digit={actual_max_digit}, expected_max_digit={expected_max_digit}"
90            ));
91        }
92
93        let message_digits =
94            (wt_params.message_byte_len() * 8).div_ceil(paramset.winternitz_log_d) as usize;
95        let checksum_digits = wt_params.total_digit_len() as usize - message_digits;
96
97        let mut elements: Vec<&[u8]> = cur_witness_iter
98            .by_ref()
99            .skip(1)
100            .step_by(2)
101            .take(message_digits)
102            .collect();
103        elements.reverse();
104
105        // advance iterator to skip checksum digits at the end
106        cur_witness_iter.by_ref().nth(checksum_digits * 2 - 1);
107
108        commits.push(
109            elements
110                .chunks_exact(2)
111                .map(|digits| {
112                    let first_digit = parse_winternitz_digit(digits[0], expected_max_digit)?;
113                    let second_digit = parse_winternitz_digit(digits[1], expected_max_digit)?;
114
115                    Ok(second_digit * digit_base + first_digit)
116                })
117                .collect::<Result<Vec<_>>>()?,
118        );
119    }
120    commits.reverse();
121    Ok(commits)
122}

124/// Extracts the Winternitz signatures of the committed data from the witness.
125///
126/// Digits are returned in canonical form (see [`canonicalize_winternitz_digit`]).
127/// Consensus accepts non-minimal and out-of-range digits, but script execution
128/// with minimal-data checks rejects them and witnesses that reuse them are
129/// non-standard.
130/// Note: The function is hardcoded for winternitz_log_d = 4 currently, will not work for others.
131pub fn extract_winternitz_commits_with_sigs(
132    witness: Witness,
133    wt_derive_paths: &[WinternitzDerivationPath],
134    paramset: &'static ProtocolParamset,
135) -> Result<Vec<Vec<Vec<u8>>>> {
136    if paramset.winternitz_log_d != 4 {
137        return Err(eyre::eyre!("Only winternitz_log_d = 4 is supported"));
138    }
139    // Structure: [commit][signature_sequence][element]
140    // - commit: one signed message
141    // - signature_sequence: alternating signature elements and signed characters, ending with a checksum
142    // - element: raw bytes of either a signature part, signed character, or checksum
143    let mut commits_with_sig: Vec<Vec<Vec<u8>>> = Vec::new();
144    let mut cur_witness_iter = witness.into_iter().skip(1);
145
146    for wt_path in wt_derive_paths.iter().rev() {
147        let wt_params = wt_path.get_params();
148        let max_digit =
149            u8::try_from(wt_params.max_digit()).wrap_err("Failed to convert max digit to u8")?;
150        let message_digits =
151            (wt_params.message_byte_len() * 8).div_ceil(paramset.winternitz_log_d) as usize;
152        let checksum_digits = wt_params.total_digit_len() as usize - message_digits;
153
154        let elements: Vec<Vec<u8>> = cur_witness_iter
155            .by_ref()
156            .take((message_digits + checksum_digits) * 2)
157            .enumerate()
158            .map(|(idx, element)| {
159                // Elements alternate between a 20-byte hash and its digit.
160                if idx % 2 == 1 {
161                    canonicalize_winternitz_digit(element, max_digit)
162                } else {
163                    Ok(element.to_vec())
164                }
165            })
166            .collect::<Result<_>>()?;
167
168        commits_with_sig.push(elements);
169    }
170
171    Ok(commits_with_sig)
172}

174/// A trait that marks all script types. Each script has a `generate_script_inputs` (eg. [`WinternitzCommit::generate_script_inputs`]) function that
175/// generates the witness for the script using various arguments. A `dyn SpendableScript` is cast into a concrete [`ScriptKind`] to
176/// generate a witness, the trait object can be used to generate the script_buf.
177///
178/// We store `Arc<dyn SpendableScript>`s inside a [`super::transaction::TxHandler`] input, and we cast them into a [`ScriptKind`] when signing.
179///
180/// When creating a new Script, make sure you add it to the [`ScriptKind`] enum and add a test for it below.
181/// Otherwise, it will not be spendable.
182pub trait SpendableScript: Send + Sync + 'static + std::any::Any {
183    fn as_any(&self) -> &dyn Any;
184
185    fn kind(&self) -> ScriptKind<'_>;
186
187    fn to_script_buf(&self) -> ScriptBuf;
188}
189
190impl Debug for dyn SpendableScript {
191    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
192        write!(f, "SpendableScript")
193    }
194}

264#[derive(Clone)]
265pub struct Multisig {
266    pubkeys: Vec<XOnlyPublicKey>,
267    threshold: u32,
268}
269
270impl SpendableScript for Multisig {
271    fn as_any(&self) -> &dyn Any {
272        self
273    }
274
275    fn kind(&self) -> ScriptKind<'_> {
276        ScriptKind::ManualSpend(self)
277    }
278
279    fn to_script_buf(&self) -> ScriptBuf {
280        let mut script_builder = Builder::new()
281            .push_x_only_key(&self.pubkeys[0])
282            .push_opcode(OP_CHECKSIG);
283        for pubkey in self.pubkeys.iter().skip(1) {
284            script_builder = script_builder.push_x_only_key(pubkey);
285            script_builder = script_builder.push_opcode(OP_CHECKSIGADD);
286        }
287        script_builder = script_builder.push_int(self.threshold as i64);
288        script_builder = script_builder.push_opcode(OP_NUMEQUAL);
289        script_builder.into_script()
290    }
291}

293impl Multisig {
294    pub fn new(pubkeys: Vec<XOnlyPublicKey>, threshold: u32) -> Self {
295        Self { pubkeys, threshold }
296    }
297
298    pub fn from_security_council(security_council: SecurityCouncil) -> Self {
299        Self {
300            pubkeys: security_council.pks,
301            threshold: security_council.threshold,
302        }
303    }
304
305    pub fn generate_script_inputs(
306        &self,
307        signatures: &[Option<taproot::Signature>],
308    ) -> eyre::Result<Witness> {
309        let mut witness = Witness::new();
310
311        for signature in signatures.iter().rev() {
312            match signature {
313                Some(sig) => witness.push(sig.serialize()),
314                None => witness.push([]),
315            }
316        }
317        Ok(witness)
318    }

321/// Struct for scripts that commit to a message using Winternitz keys
322/// Contains the Winternitz PK, CheckSig PK, message length respectively
323/// can contain multiple different Winternitz public keys for different messages
324#[derive(Clone)]
325pub struct WinternitzCommit {
326    commitments: Vec<(PublicKey, u32)>,
327    pub(crate) checksig_pubkey: XOnlyPublicKey,
328    log_d: u32,
329}
330
331impl SpendableScript for WinternitzCommit {
332    fn as_any(&self) -> &dyn Any {
333        self
334    }
335
336    fn kind(&self) -> ScriptKind<'_> {
337        ScriptKind::WinternitzCommit(self)
338    }
339
340    fn to_script_buf(&self) -> ScriptBuf {
341        let mut total_script = ScriptBuf::new();
342        for (index, (pubkey, _size)) in self.commitments.iter().enumerate() {
343            let params = self.get_params(index);
344            let a = bitvm::signatures::signing_winternitz::WINTERNITZ_MESSAGE_VERIFIER
345                .checksig_verify_and_clear_stack(&params, pubkey);
346            total_script.extend(a.compile().instructions().map(|x| x.expect("just created")));
347        }
348
349        total_script.push_slice(self.checksig_pubkey.serialize());
350        total_script.push_opcode(OP_CHECKSIG);
351        total_script
352    }
353}
354
355impl WinternitzCommit {
356    pub fn get_params(&self, index: usize) -> Parameters {
357        Parameters::new(self.commitments[index].1, self.log_d)
358    }
359
360    pub fn generate_script_inputs(
361        &self,
362        commit_data: &[(Vec<u8>, SecretKey)],
363        signature: &taproot::Signature,
364    ) -> Witness {
365        let mut witness = Witness::new();
366        witness.push(signature.serialize());
367        for (index, (data, secret_key)) in commit_data.iter().enumerate().rev() {
368            #[cfg(debug_assertions)]
369            {
370                let pk = bitvm::signatures::winternitz::generate_public_key(
371                    &self.get_params(index),
372                    secret_key,
373                );
374                if pk != self.commitments[index].0 {
375                    tracing::error!(
376                        "Winternitz public key mismatch len: {}",
377                        self.commitments[index].1
378                    );
379                }
380            }
381            bitvm::signatures::signing_winternitz::WINTERNITZ_MESSAGE_VERIFIER
382                .sign(&self.get_params(index), secret_key, data)
383                .into_iter()
384                .for_each(|x| witness.push(x));
385        }
386        witness
387    }
388
389    /// commitments is a Vec of winternitz public key and message length tuple
390    pub fn new(
391        commitments: Vec<(PublicKey, u32)>,
392        checksig_pubkey: XOnlyPublicKey,
393        log_d: u32,
394    ) -> Self {
395        Self {
396            commitments,
397            checksig_pubkey,
398            log_d,
399        }
400    }
401}

403/// Struct for scripts that include a relative timelock (by block count) and optionally a CHECKSIG if a pubkey is provided.
404/// Generates a relative timelock script with a given [`XOnlyPublicKey`] that CHECKSIG checks the signature against.
405///
406/// ATTENTION: If you want to spend a UTXO using timelock script, the
407/// condition is that (`# in the script`) ≤ (`# in the sequence of the tx`)
408/// ≤ (`# of blocks mined after UTXO appears on the chain`). However, this is not mandatory.
409/// One can spend an output delayed for some number of blocks just by using the nSequence field
410/// of the input inside the transaction. For more, see:
411///
412/// - [BIP-0068](https://github.com/bitcoin/bips/blob/master/bip-0068.mediawiki)
413/// - [BIP-0112](https://github.com/bitcoin/bips/blob/master/bip-0112.mediawiki)
414#[derive(Debug, Clone)]
415pub struct TimelockScript(pub(crate) Option<XOnlyPublicKey>, u16);
416
417impl SpendableScript for TimelockScript {
418    fn as_any(&self) -> &dyn Any {
419        self
420    }
421
422    fn kind(&self) -> ScriptKind<'_> {
423        ScriptKind::TimelockScript(self)
424    }
425
426    fn to_script_buf(&self) -> ScriptBuf {
427        let script_builder = Builder::new()
428            .push_int(self.1 as i64)
429            .push_opcode(OP_CSV)
430            .push_opcode(OP_DROP);
431
432        if let Some(xonly_pk) = self.0 {
433            script_builder
434                .push_x_only_key(&xonly_pk)
435                .push_opcode(OP_CHECKSIG)
436        } else {
437            script_builder.push_opcode(OP_TRUE)
438        }
439        .into_script()
440    }
441}
442
443impl TimelockScript {
444    pub fn generate_script_inputs(&self, signature: Option<&taproot::Signature>) -> Witness {
445        match signature {
446            Some(sig) => Witness::from_slice(&[sig.serialize()]),
447            None => Witness::default(),
448        }
449    }
450
451    pub fn new(xonly_pk: Option<XOnlyPublicKey>, block_count: u16) -> Self {
452        Self(xonly_pk, block_count)
453    }
454}

456/// Struct for scripts that reveal a preimage of a OP_HASH160 and verify it against the given hash in the script.
457pub struct PreimageRevealScript(pub(crate) XOnlyPublicKey, [u8; 20]);
458
459impl SpendableScript for PreimageRevealScript {
460    fn as_any(&self) -> &dyn Any {
461        self
462    }
463
464    fn kind(&self) -> ScriptKind<'_> {
465        ScriptKind::PreimageRevealScript(self)
466    }
467
468    fn to_script_buf(&self) -> ScriptBuf {
469        Builder::new()
470            .push_opcode(OP_SIZE)
471            .push_int(20)
472            .push_opcode(OP_EQUALVERIFY)
473            .push_opcode(OP_HASH160)
474            .push_slice(self.1)
475            .push_opcode(OP_EQUALVERIFY)
476            .push_x_only_key(&self.0)
477            .push_opcode(OP_CHECKSIG)
478            .into_script()
479    }
480}

482impl PreimageRevealScript {
483    pub fn generate_script_inputs(
484        &self,
485        preimage: impl AsRef<[u8]>,
486        signature: &taproot::Signature,
487    ) -> Witness {
488        let mut witness = Witness::new();
489        #[cfg(debug_assertions)]
490        assert_eq!(
491            bitcoin::hashes::hash160::Hash::hash(preimage.as_ref()),
492            bitcoin::hashes::hash160::Hash::from_byte_array(self.1),
493            "Preimage does not match"
494        );
495
496        witness.push(signature.serialize());
497        witness.push(preimage.as_ref());
498        witness
499    }
500
501    pub fn new(xonly_pk: XOnlyPublicKey, hash: [u8; 20]) -> Self {
502        Self(xonly_pk, hash)
503    }

781                        Sequence::from_height(seq)
782                    }
783                    _ => DEFAULT_SEQUENCE,
784                }
785            } else {
786                DEFAULT_SEQUENCE
787            }
788        } else {
789            DEFAULT_SEQUENCE
790        };

792        builder = builder.add_input(
793            crate::rpc::clementine::NormalSignatureKind::OperatorSighashDefault,
794            SpendableTxIn::new(
795                outpoint,
796                TxOut {
797                    value: amount,
798                    script_pubkey: address.script_pubkey(),
799                },
800                scripts.clone(),
801                Some(taproot_spend_info.clone()),
802            ),
803            spend_path,
804            sequence,
805        );

810                script_pubkey: address.script_pubkey(),

1268            extract_winternitz_commits(witness.clone(), std::slice::from_ref(&kickoff), paramset)
1269                .unwrap();
1270        assert_eq!(
1271            normal_extracted[0], original_message,
1272            "Normal extraction should match original message"
1273        );
1274
1275        // Witness layout: [taproot_sig, hash0, digit0, hash1, digit1, ...]
1276        // Digits are at indices 2, 4, 6, ... (0-indexed)
1277        //
1278        // The digits are in forward order: digit0 is for the first position in the
1279        // Winternitz signature. After extraction, elements are reversed and chunked
1280        // into pairs: [high_nibble, low_nibble] → byte = high * 16 + low.
1281        //
1282        // Find a message digit element (at even witness index >= 2) with value 15.
1283        // Digit elements are at indices 2, 4, 6, ... (skipping hash elements at 1, 3, 5, ...).
1284        let witness_vec: Vec<Vec<u8>> = witness.to_vec();
1285        let message_digits =
1286            (kickoff.get_params().message_byte_len() * 8).div_ceil(paramset.winternitz_log_d);
1287        let mut tampered_idx = None;
1288        for i in 0..message_digits as usize {
1289            let witness_idx = 2 + 2 * i;
1290            if witness_vec[witness_idx] == vec![0x0F] {
1291                tampered_idx = Some(witness_idx);
1292                break;
1293            }
1294        }
1295        let tampered_idx = tampered_idx.expect("Should find a digit element with value 15");
1296
1297        let mut tampered_witness_vec = witness_vec.clone();
1298
1299        // Change digit from 15 to 256. On-chain OP_MIN clamps it back to 15.
1300        let mut scriptint_buf = [0u8; 8];
1301        let scriptint_len = bitcoin::script::write_scriptint(&mut scriptint_buf, 256);
1302        tampered_witness_vec[tampered_idx] = scriptint_buf[..scriptint_len].to_vec();
1303
1304        let tampered_witness = bitcoin::Witness::from_slice(&tampered_witness_vec);
1305
1306        // Extract from tampered witness
1307        let tampered_extracted = extract_winternitz_commits(
1308            tampered_witness.clone(),
1309            std::slice::from_ref(&kickoff),
1310            paramset,
1311        )
1312        .unwrap();
1313
1314        // The extracted data should match what on-chain verification validated after OP_MIN.
1315        assert_eq!(
1316            tampered_extracted[0], original_message,
1317            "Extraction should match original message
1318             Original: {original_message:?}, Tampered extraction: {:?}",
1319            tampered_extracted[0]
1320        );
1321
1322        let mut non_minimal_witness_vec = witness_vec.clone();
1323
1324        // Change digit from 15 to a non-minimal script integer encoding of
1325        // 256. Without SCRIPT_VERIFY_MINIMALDATA, script arithmetic decodes
1326        // this as 256 and OP_MIN clamps it back to 15.
1327        non_minimal_witness_vec[tampered_idx] = vec![0x00, 0x01, 0x00];
1328
1329        let non_minimal_witness = bitcoin::Witness::from_slice(&non_minimal_witness_vec);
1330
1331        let non_minimal_extracted = extract_winternitz_commits(
1332            non_minimal_witness.clone(),
1333            std::slice::from_ref(&kickoff),
1334            paramset,
1335        )
1336        .unwrap();
1337
1338        // SCRIPT_VERIFY_MINIMALDATA is a standardness/policy rule here, not a
1339        // mandatory consensus rule. A transaction mined directly into a block
1340        // can reveal this non-minimal digit, so off-chain extraction must mirror
1341        // consensus script arithmetic: decode 256, then clamp it to 15.
1342        assert_eq!(
1343            non_minimal_extracted[0], original_message,
1344            "Extraction should match original message
1345             Original: {original_message:?}, Non-minimal extraction: {:?}",
1346            non_minimal_extracted[0]
1347        );
1348
1349        // ===== Verify on-chain script ACCEPTS the tampered witness =====
1350        //
1351        // Build the on-chain Winternitz verification script and run it with the
1352        // oversized but minimally encoded witness to prove that OP_MIN clamps it.
1353        // The BitVM script executor enables minimal-data policy by default, so it
1354        // cannot be used to prove the non-minimal witness case above.
1355        use bitvm::signatures::winternitz::{
1356            generate_public_key as wt_generate_public_key, ListpickVerifier, VoidConverter,
1357            Winternitz,
1358        };
1359
1360        let wt_params = kickoff.get_params();
1361        let wt_sk = signer.get_derived_winternitz_sk(kickoff.clone()).unwrap();
1362        let wt_pk = wt_generate_public_key(&wt_params, &wt_sk);
1363
1364        let verifier = Winternitz::<ListpickVerifier, VoidConverter>::new();
1365        let verify_script = verifier.checksig_verify_and_clear_stack(&wt_params, &wt_pk);
1366
1367        // Build complete script: checksig_verify_and_clear_stack + OP_TRUE
1368        let mut script_bytes = verify_script.compile().to_bytes();
1369        script_bytes.push(0x51); // OP_TRUE
1370
1371        // The signing witness is the tampered witness without the taproot signature prefix
1372        let signing_witness: Vec<Vec<u8>> = tampered_witness_vec[1..].to_vec();
1373
1374        let result = bitvm::execute_raw_script_with_inputs(script_bytes.clone(), signing_witness);
1375        assert!(
1376            result.success,
1377            "On-chain Winternitz verification should ACCEPT the tampered witness \
1378             (digit=256 is clamped to 15 by OP_MIN). Script error: {:?}",
1379            result.error
1380        );
1381
1382        // The minimal-data executor rejects the non-minimal digit. Verifiers
1383        // use this executor to find additional disprove witnesses.
1384        let result = bitvm::execute_raw_script_with_inputs(
1385            script_bytes.clone(),
1386            non_minimal_witness_vec[1..].to_vec(),
1387        );
1388        assert!(
1389            !result.success,
1390            "Minimal-data execution should reject the non-minimal digit"
1391        );
1392
1393        // Signatures extracted for disprove witnesses use the honest encoding.
1394        for witness in [tampered_witness, non_minimal_witness] {
1395            let canonical_signature = extract_winternitz_commits_with_sigs(
1396                witness,
1397                std::slice::from_ref(&kickoff),
1398                paramset,
1399            )
1400            .unwrap()
1401            .remove(0);
1402            assert_eq!(canonical_signature, witness_vec[1..]);
1403
1404            let result =
1405                bitvm::execute_raw_script_with_inputs(script_bytes.clone(), canonical_signature);
1406            assert!(
1407                result.success,
1408                "Canonical witness should pass minimal-data execution. Script error: {:?}",
1409                result.error
1410            );
1411        }
1412
1413        // Negative zero is a non-minimal encoding of digit 0.
1414        assert_eq!(
1415            canonicalize_winternitz_digit(&[0x80], 15).unwrap(),
1416            Vec::<u8>::new()
1417        );
1418    }
1419}
