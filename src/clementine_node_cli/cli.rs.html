31/// Clementine CLI for node operators. Not intended for end users of the bridge.
32#[derive(Parser)]
33#[command(author, version, about, long_about = None)]
34struct Cli {
35    /// The URL of the service
36    #[arg(short, long)]
37    node_url: String,

39    #[command(subcommand)]
40    command: Commands,

224fn get_path_from_env_or_default(env_var: &str, default: &str) -> PathBuf {
225    let path = std::env::var(env_var);
226    let path = match path {
227        Ok(path) => {
228            println!("Using cert path from environment variable {env_var}: {path}");
229            path
230        }
231        Err(_) => {
232            println!("Warning: {env_var} is not set, using default cert path: {default}.\nIf this path is incorrect, please set the environment variable {env_var} to the correct path or call the binary from the correct directory, or any aggregator/operator/verifier command may not work.");
233            default.to_string()
234        }
235    };
236    PathBuf::from(path)
237}

239// Create a minimal config with default TLS paths
240fn create_minimal_config() -> BridgeConfig {
241    // CLIENT_KEY_PATH env var will be used if it is set
242    // CLIENT_CERT_PATH env var will be used if it is set
243    // CA_CERT_PATH env var will be used if it is set
244    BridgeConfig {
245        ca_cert_path: get_path_from_env_or_default("CA_CERT_PATH", "core/certs/ca/ca.pem"),
246        client_cert_path: get_path_from_env_or_default(
247            "CLIENT_CERT_PATH",
248            "core/certs/client/client.pem",
249        ),
250        client_key_path: get_path_from_env_or_default(
251            "CLIENT_KEY_PATH",
252            "core/certs/client/client.key",
253        ),
254        ..Default::default()
255    }

258fn parse_script_buf(script: &str) -> Result<ScriptBuf, String> {
259    ScriptBuf::from_hex(script).map_err(|error| error.to_string())

262fn parse_evm_address(address: &str) -> Result<EVMAddress, String> {
263    alloy::primitives::Address::from_str(address)
264        .map(|address| EVMAddress(address.into_array()))
265        .map_err(|error| error.to_string())
266}

291async fn handle_operator_call(url: String, command: OperatorCommands) {
292    let config = create_minimal_config();
293    let mut operator = clementine_core::rpc::get_clients(
294        vec![url],
295        clementine_core::rpc::operator_client_builder(&config),
296        &config,
297        true,
298    )
299    .await
300    .expect("Exists")[0]
301        .clone();
302
303    match command {
304        OperatorCommands::GetParams => {
305            let params = operator
306                .get_params(Empty {})
307                .await
308                .expect("Failed to make a request to operator");
309            println!("Operator params: {params:?}");
310        }
311        OperatorCommands::Withdraw { withdrawal } => {
312            println!("Processing withdrawal with id {}", withdrawal.withdrawal_id);
313
314            operator
315                .internal_withdraw(Request::new(withdrawal_params_from_args(withdrawal)))
316                .await
317                .expect("Failed to make a request to operator");
318        }
319        OperatorCommands::Vergen => {
320            let params = Empty {};
321            let response = operator
322                .vergen(Request::new(params))
323                .await
324                .expect("Failed to make a request to operator");
325            let response_msg = response.into_inner().response;
326            println!("Vergen response:\n{response_msg}");
327        }
328        OperatorCommands::GetReimbursementTxs {
329            deposit_outpoint_txid,
330            deposit_outpoint_vout,
331        } => {
332            println!(
333                "Getting kickoff txs for outpoint {deposit_outpoint_txid}:{deposit_outpoint_vout}"
334            );
335            let response = operator
336                .get_reimbursement_txs(Request::new(rpc_outpoint(
337                    deposit_outpoint_txid,
338                    deposit_outpoint_vout,
339                )))
340                .await
341                .expect("Failed to make a request to operator")
342                .into_inner();
343            for signed_tx in &response.signed_txs {
344                let tx_type: TransactionType = signed_tx
345                    .transaction_type
346                    .expect("Tx type should not be None")
347                    .try_into()
348                    .expect("Failed to convert tx type");
349                let transaction: bitcoin::Transaction =
350                    bitcoin::consensus::deserialize(&signed_tx.raw_tx)
351                        .expect("Failed to decode transaction");
352                match tx_type {
353                    TransactionType::Kickoff => {
354                        println!("Round tx is on chain, time to send the kickoff tx. This tx is non-standard and cannot be sent by using normal Bitcoin RPC");
355                    }
356                    TransactionType::BurnUnusedKickoffConnectors => {
357                        println!("To be able to send ready to reimburse tx, all unused kickoff connectors must be burned, otherwise the operator will get slashed.
358                        This tx is standard and requires CPFP to be sent (last output is the anchor output)");
359                    }
360                    TransactionType::ReadyToReimburse => {
361                        println!("All unused kickoff connectors are burned, and all live kickoffs kickoff finalizer utxo's are
362                        spent, meaning it is safe to send ready to reimburse tx. This tx is standard and requires CPFP to be sent (last output is the anchor output)");
363                    }
364                    TransactionType::Reimburse => {
365                        println!("Reimburse tx is ready to be sent. This tx is standard and requires CPFP to be sent (last output is the anchor output)");
366                    }
367                    TransactionType::ChallengeTimeout => {
368                        println!("After kickoff, challenge timeout tx needs to be sent. Due to the timelock, it can only be sent after 216 blocks pass from the kickoff tx {}.
369                        This tx is standard and requires CPFP to be sent (last output is the anchor output)",
370                        transaction.input[0].previous_output.txid);
371                    }
372                    TransactionType::Round => {
373                        println!("Time to send the round tx either for sending the kickoff tx, or getting the reimbursement for the past kickoff by advancing the round. Round tx is a non-standard tx and cannot be sent by using normal Bitcoin RPC.
374                        If the round is not the first round, 216 number of blocks need to pass from the previous ready to reimburse tx {} (If this is not collateral)",
375                        transaction.input[0].previous_output.txid);
376                    }
377                    _ => {}
378                }
379                let hex_tx = hex::encode(&signed_tx.raw_tx);
380                println!("Tx type: {tx_type:?}, Tx hex: {hex_tx:?}");
381            }
382        }
383        OperatorCommands::GetCompatibilityParams => {
384            let params = operator
385                .get_compatibility_params(Empty {})
386                .await
387                .expect("Failed to make a request");
388            println!("Compatibility params:\n{params:#?}");
389        }
390        OperatorCommands::GetEntityStatus => {
391            let params = operator
392                .get_current_status(Empty {})
393                .await
394                .expect("Failed to make a request");
395            println!("Entity status:\n{params:#?}");
396        }
397    }
398}

447async fn handle_aggregator_call(url: String, command: AggregatorCommands) {
448    println!("Connecting to aggregator at {url}");
449    let config = create_minimal_config();
450    let mut aggregator = clementine_core::rpc::get_clients(
451        vec![url],
452        ClementineAggregatorClient::new,
453        &config,
454        false,
455    )
456    .await
457    .expect("Exists")[0]
458        .clone();
459
460    match command {
461        AggregatorCommands::Setup => {
462            let setup = aggregator
463                .setup(Empty {})
464                .await
465                .expect("Failed to make a request");
466            println!("{setup:?}");
467        }
468        AggregatorCommands::GetCompatibilityParamsFromAll => {
469            let params = aggregator
470                .get_compatibility_data_from_entities(Empty {})
471                .await
472                .expect("Failed to make a request");
473            let params = params.into_inner();
474            println!("Compatibility params from all entities:");
475            for entity in params.entities_compatibility_data {
476                match entity.entity_id {
477                    Some(entity_id) => {
478                        let kind = EntityType::try_from(entity_id.kind)
479                            .expect("Failed to convert kind to entity type");
480                        println!("Entity: {:?}, ID: {:?}", kind, entity_id.id);
481                    }
482                    None => {
483                        println!("No entity id received");
484                    }
485                }
486                match entity.data_result {
487                    Some(data_result) => match data_result {
488                        DataResult::Data(data) => {
489                            println!("{data:#?}");
490                        }
491                        DataResult::Error(error) => {
492                            println!("Error: {error}");
493                        }
494                    },
495                    None => {
496                        println!("No data");
497                    }
498                }
499            }
500        }
501        AggregatorCommands::NewDeposit {
502            deposit_outpoint_txid,
503            deposit_outpoint_vout,
504            evm_address,
505            recovery_taproot_address,
506        } => {
507            let deposit_outpoint = rpc_outpoint(deposit_outpoint_txid, deposit_outpoint_vout);
508
509            let move_to_vault_tx = aggregator
510                .new_deposit(Deposit {
511                    deposit_outpoint: Some(deposit_outpoint.clone()),
512                    deposit_data: Some(DepositData::BaseDeposit(BaseDeposit {
513                        evm_address: evm_address.0.to_vec(),
514                        recovery_taproot_address: recovery_taproot_address
515                            .assume_checked()
516                            .to_string(),
517                    })),
518                })
519                .await
520                .expect("Failed to make a request");
521
522            let move_to_vault_tx = move_to_vault_tx.into_inner();
523
524            let deposit = aggregator
525                .send_move_to_vault_tx(SendMoveTxRequest {
526                    raw_tx: Some(move_to_vault_tx.clone()),
527                    deposit_outpoint: Some(deposit_outpoint),
528                })
529                .await;
530
531            match deposit {
532                Ok(deposit) => {
533                    let move_txid = deposit.get_ref().txid.clone();
534                    let txid = bitcoin::Txid::from_byte_array(
535                        move_txid
536                            .try_into()
537                            .expect("Failed to convert txid to array"),
538                    );
539                    println!("Move txid: {txid}");
540                }
541                Err(e) => {
542                    println!("Failed to send move transaction: {e}");
543                    println!(
544                        "Please send manually: {}",
545                        hex::encode(move_to_vault_tx.raw_tx)
546                    );
547                }
548            }
549        }
550        AggregatorCommands::NewOptimisticWithdrawal {
551            withdrawal,
552            verification_signature,
553        } => {
554            println!("Processing withdrawal with id {}", withdrawal.withdrawal_id);
555
556            let withdraw_params_with_sig =
557                clementine_core::rpc::clementine::OptimisticWithdrawParams {
558                    withdrawal: Some(withdrawal_params_from_args(withdrawal)),
559                    verification_signature: verification_signature.clone(),
560                };
561
562            let response = aggregator
563                .optimistic_payout(Request::new(withdraw_params_with_sig))
564                .await
565                .expect("Failed to make a request");
566            println!("Tx: {}", hex::encode(response.get_ref().raw_tx.clone()));
567        }
568        AggregatorCommands::GetNofnAggregatedKey => {
569            let response = aggregator
570                .get_nofn_aggregated_xonly_pk(Request::new(Empty {}))
571                .await
572                .expect("Failed to make a request");
573            let xonly_pk = bitcoin::XOnlyPublicKey::from_slice(&response.get_ref().nofn_xonly_pk)
574                .expect("Failed to parse xonly_pk");
575            println!("{xonly_pk}");
576        }
577        AggregatorCommands::GetDepositAddress {
578            evm_address,
579            recovery_taproot_address,
580            network,
581            user_takes_after,
582        } => {
583            let response = aggregator
584                .get_nofn_aggregated_xonly_pk(Request::new(Empty {}))
585                .await
586                .expect("Failed to make a request");
587            let xonly_pk = bitcoin::XOnlyPublicKey::from_slice(&response.get_ref().nofn_xonly_pk)
588                .expect("Failed to parse xonly_pk");
589
590            let deposit_address = clementine_core::builder::address::generate_deposit_address(
591                xonly_pk,
592                &recovery_taproot_address,
593                evm_address,
594                network,
595                user_takes_after,
596            )
597            .expect("Failed to generate deposit address");
598
599            let address = &deposit_address.0;
600            println!("Deposit address: {address}");
601        }
602        AggregatorCommands::InternalGetEmergencyStopTx { move_txids } => {
603            let emergency_stop_tx = aggregator
604                .internal_get_emergency_stop_tx(Request::new(
605                    clementine::GetEmergencyStopTxRequest {
606                        txids: move_txids.clone().into_iter().map(Into::into).collect(),
607                    },
608                ))
609                .await
610                .expect("Failed to make a request");
611            println!("Emergency stop tx: {emergency_stop_tx:?}");
612            let response = emergency_stop_tx.into_inner();
613            let mut returned_txids = Vec::new();
614            for (txid, tx) in response
615                .txids
616                .into_iter()
617                .zip(response.encrypted_emergency_stop_txs)
618            {
619                let txid = Txid::try_from(txid).expect("Failed to parse move txid");
620                println!("Emergency stop tx for move tx {txid}: {}", hex::encode(tx));
621                returned_txids.push(txid);
622            }
623            for txid in move_txids
624                .iter()
625                .filter(|txid| !returned_txids.contains(txid))
626            {
627                println!("No emergency stop tx stored for move tx {txid}");
628            }
629        }
630        AggregatorCommands::GetReplacementDepositAddress {
631            move_txid,
632            network,
633            security_council,
634        } => {
635            let response = aggregator
636                .get_nofn_aggregated_xonly_pk(Request::new(Empty {}))
637                .await
638                .expect("Failed to make a request");
639
640            let nofn_xonly_pk =
641                bitcoin::XOnlyPublicKey::from_slice(&response.get_ref().nofn_xonly_pk)
642                    .expect("Failed to parse xonly_pk");
643
644            let (replacement_deposit_address, _) =
645                clementine_core::builder::address::generate_replacement_deposit_address(
646                    move_txid,
647                    nofn_xonly_pk,
648                    network,
649                    security_council,
650                )
651                .expect("Failed to generate replacement deposit address");
652
653            println!("Replacement deposit address: {replacement_deposit_address}");
654        }
655        AggregatorCommands::NewReplacementDeposit {
656            deposit_outpoint_txid,
657            deposit_outpoint_vout,
658            old_move_txid,
659        } => {
660            let deposit_outpoint = rpc_outpoint(deposit_outpoint_txid, deposit_outpoint_vout);
661
662            let deposit = aggregator
663                .new_deposit(Deposit {
664                    deposit_outpoint: Some(deposit_outpoint.clone()),
665                    deposit_data: Some(DepositData::ReplacementDeposit(ReplacementDeposit {
666                        old_move_txid: Some(old_move_txid.into()),
667                    })),
668                })
669                .await
670                .expect("Failed to make a request");
671            let deposit = aggregator
672                .send_move_to_vault_tx(SendMoveTxRequest {
673                    raw_tx: Some(deposit.into_inner()),
674                    deposit_outpoint: Some(deposit_outpoint),
675                })
676                .await
677                .expect("Failed to make a request");
678            let move_txid = deposit.get_ref().txid.clone();
679            let txid = bitcoin::Txid::from_byte_array(
680                move_txid
681                    .try_into()
682                    .expect("Failed to convert txid to array"),
683            );
684            println!("Move txid: {txid}");
685        }
686        AggregatorCommands::NewWithdrawal {
687            withdrawal,
688            verification_signature,
689            operator_xonly_pks,
690        } => {
691            println!("Processing withdrawal with id {}", withdrawal.withdrawal_id);
692
693            let withdraw_params_with_sig =
694                clementine_core::rpc::clementine::WithdrawParamsWithSig {
695                    withdrawal: Some(withdrawal_params_from_args(withdrawal)),
696                    verification_signature,
697                };
698
699            let operator_xonly_pks = operator_xonly_pks
700                .unwrap_or_default()
701                .into_iter()
702                .map(XOnlyPublicKeyRpc::from)
703                .collect::<Vec<_>>();
704
705            let response = aggregator
706                .withdraw(Request::new(AggregatorWithdrawalInput {
707                    withdrawal: Some(withdraw_params_with_sig),
708                    operator_xonly_pks,
709                }))
710                .await
711                .expect("Failed to make a request");
712
713            let withdraw_responses = response.get_ref().withdraw_responses.clone();
714
715            for (i, result) in withdraw_responses.iter().enumerate() {
716                println!("Operator {i}: {result:?}");
717            }
718        }
719        AggregatorCommands::GetEntityStatuses { restart_tasks } => {
720            let restart_tasks = restart_tasks.unwrap_or(false);
721            let request = GetEntityStatusesRequest { restart_tasks };
722
723            let response = aggregator
724                .get_entity_statuses(Request::new(request))
725                .await
726                .expect("Failed to make a request");
727
728            println!("Entities status:");
729            for entity_status in &response.get_ref().entity_statuses {
730                match &entity_status.entity_id {
731                    Some(entity_id) => {
732                        let kind = EntityType::try_from(entity_id.kind)
733                            .expect("Failed to convert kind to entity type");
734                        println!("Entity: {:?}, ID: {:?}", kind, entity_id.id);
735                        match &entity_status.status_result {
736                            Some(clementine_core::rpc::clementine::entity_status_with_id::StatusResult::Status(status)) => {
737                                let EntityStatus {
738                                    automation,
739                                    wallet_balance,
740                                    tx_sender_synced_height,
741                                    finalized_synced_height,
742                                    hcp_last_proven_height,
743                                    rpc_tip_height,
744                                    bitcoin_syncer_synced_height,
745                                    state_manager_next_height,
746                                    stopped_tasks,
747                                    btc_fee_rate_sat_vb,
748                                    lcp_synced_height,
749                                } = &status;
750                                println!("  Automation: {automation}");
751                                let wallet_balance = wallet_balance
752                                    .as_ref()
753                                    .map_or("N/A".to_string(), |s| s.clone());
754                                println!("  Wallet balance: {wallet_balance}");
755                                let btc_fee_rate_sat_vb = btc_fee_rate_sat_vb
756                                    .map_or("N/A".to_string(), |r| r.to_string());
757                                println!("  BTC fee rate: {btc_fee_rate_sat_vb} sat/vB");
758                                let tx_sender_height = tx_sender_synced_height
759                                    .map_or("N/A".to_string(), |h| h.to_string());
760                                println!("  TX sender synced height: {tx_sender_height}");
761                                let finalized_height = finalized_synced_height
762                                    .map_or("N/A".to_string(), |h| h.to_string());
763                                println!("  Finalized synced height: {finalized_height}");
764                                let hcp_height = hcp_last_proven_height
765                                    .map_or("N/A".to_string(), |h| h.to_string());
766                                println!("  HCP last proven height: {hcp_height}");
767                                let rpc_tip_height = rpc_tip_height
768                                    .map_or("N/A".to_string(), |h| h.to_string());
769                                println!("  RPC tip height: {rpc_tip_height}");
770                                let bitcoin_syncer_height = bitcoin_syncer_synced_height
771                                    .map_or("N/A".to_string(), |h| h.to_string());
772                                println!("  Bitcoin syncer synced height: {bitcoin_syncer_height}");
773                                let state_manager_height = state_manager_next_height
774                                    .map_or("N/A".to_string(), |h| h.to_string());
775                                println!("  State manager next height: {state_manager_height}");
776                                let lcp_height = lcp_synced_height
777                                    .map_or("N/A".to_string(), |h| h.to_string());
778                                println!("  LCP synced height: {lcp_height}");
779                                if !stopped_tasks.as_ref().is_none_or(|t| t.stopped_tasks.is_empty()) {
780                                    let stopped_tasks = &stopped_tasks
781                                        .as_ref()
782                                        .expect("Stopped tasks are required")
783                                        .stopped_tasks;
784                                    println!("  Stopped tasks: {stopped_tasks:?}");
785                                }
786                            }
787                            Some(clementine_core::rpc::clementine::entity_status_with_id::StatusResult::Err(error)) => {
788                                let error_msg = &error.error;
789                                println!("  Error: {error_msg}");
790                            }
791                            None => {
792                                println!("  No status available");
793                            }
794                        }
795                    }
796                    None => {
797                        println!("Entity: Unknown");
798                    }
799                }
800                println!();
801            }
802
803            if restart_tasks {
804                println!("Tasks restart was requested and included in the request.");
805            }
806        }
807        AggregatorCommands::Vergen => {
808            let params = Empty {};
809            let response = aggregator
810                .vergen(Request::new(params))
811                .await
812                .expect("Failed to make a request");
813            let response_msg = response.into_inner().response;
814            println!("Vergen response:\n{response_msg}");
815        }
816    }

819async fn handle_print_addresses() {
820    // Get secret key from environment
821    let secret_key = match std::env::var("SECRET_KEY") {
822        Ok(key) => SecretKey::from_str(&key).expect("Failed to parse secret key"),
823        Err(_) => {
824            println!("Error: SECRET_KEY environment variable not set");
825            return;
826        }
827    };
828
829    // Get Bitcoin RPC credentials from environment
830    let bitcoin_rpc_url = match std::env::var("BITCOIN_RPC_URL") {
831        Ok(url) => url,
832        Err(_) => {
833            println!("Error: BITCOIN_RPC_URL environment variable not set");
834            return;
835        }
836    };
837    let bitcoin_rpc_user = match std::env::var("BITCOIN_RPC_USER") {
838        Ok(user) => user,
839        Err(_) => {
840            println!("Error: BITCOIN_RPC_USER environment variable not set");
841            return;
842        }
843    };
844    let bitcoin_rpc_password = match std::env::var("BITCOIN_RPC_PASSWORD") {
845        Ok(password) => password,
846        Err(_) => {
847            println!("Error: BITCOIN_RPC_PASSWORD environment variable not set");
848            return;
849        }
850    };
851
852    // Get network from environment or default to regtest
853    let network = match std::env::var("NETWORK") {
854        Ok(network) => Network::from_str(&network).unwrap_or(Network::Regtest),
855        Err(_) => Network::Regtest,
856    };
857
858    let actor = Actor::new(secret_key, network);
859    let taproot_address = actor.address;
860    println!("Actor's taproot address: {taproot_address}");
861
862    // Connect to Bitcoin RPC and get new address
863    let rpc = match Client::new(
864        &bitcoin_rpc_url,
865        Auth::UserPass(bitcoin_rpc_user, bitcoin_rpc_password),
866    )
867    .await
868    {
869        Ok(client) => client,
870        Err(e) => {
871            println!("Error connecting to Bitcoin RPC: {e}");
872            return;
873        }
874    };
875
876    match rpc
877        .get_new_address(None, Some(bitcoincore_rpc::json::AddressType::Bech32m))
878        .await
879    {
880        Ok(address) => {
881            let addr = address.assume_checked();
882            println!("Bitcoin wallet's new address: {addr}");
883        }
884        Err(e) => println!("Error getting new address from Bitcoin wallet: {e}"),
885    }
886}

888async fn handle_bitcoin_call(url: String, command: BitcoinCommands) {
889    match command {
890        BitcoinCommands::SendTxWithCpfp {
891            raw_tx,
892            fee_payer_address,
893            fee_rate,
894            bitcoin_rpc_user,
895            bitcoin_rpc_password,
896        } => {
897            let tx_hex = hex::decode(raw_tx).expect("Failed to decode transaction");
898            let tx: bitcoin::Transaction = bitcoin::consensus::deserialize(&tx_hex)
899                .expect("Failed to deserialize transaction");
900
901            let txid = tx.compute_txid();
902            println!("Transaction created: {txid}");
903            let raw_tx = hex::encode(tx_hex);
904            println!("Raw transaction: {raw_tx}");
905
906            // Find P2A anchor output (script: 51024e73)
907            let p2a_vout = tx
908                .output
909                .iter()
910                .position(|output| {
911                    output.script_pubkey == ScriptBuf::from_hex("51024e73").expect("valid script")
912                })
913                .expect("P2A anchor output not found in transaction");
914
915            let p2a_txout = tx.output[p2a_vout].clone();
916
917            println!("Found P2A anchor output at vout: {p2a_vout}");
918
919            // Connect to Bitcoin RPC
920            use bitcoincore_rpc::{Auth, Client, RpcApi};
921            let rpc = Client::new(&url, Auth::UserPass(bitcoin_rpc_user, bitcoin_rpc_password))
922                .await
923                .expect("Failed to connect to Bitcoin RPC");
924
925            let Some(fee_payer_address) = fee_payer_address else {
926                let temp_address = rpc
927                    .get_new_address(
928                        Some("fee_payer_address"),
929                        Some(bitcoincore_rpc::json::AddressType::Bech32m),
930                    )
931                    .await
932                    .expect("Failed to get new address");
933                println!(
934                    "You haven't provided a fee payer address, so a new one was generated: {}",
935                    temp_address.assume_checked()
936                );
937                println!("Please use this address for the fee payer in the next command");
938                return;
939            };
940
941            let fee_payer_address = fee_payer_address.assume_checked();
942
943            let fee_rate_sat_vb = fee_rate.unwrap_or(10.0) as u64;
944
945            // Calculate package fee requirements
946            let parent_weight = tx.weight();
947            // empirical - tx with 1 anchor, 1 taproot input + 1 change taproot output had 540 WU
948            let estimated_child_weight = bitcoin::Weight::from_wu(540);
949            let total_weight = parent_weight + estimated_child_weight;
950            let required_fee_sats =
951                (total_weight.to_wu() as f64 * fee_rate_sat_vb as f64 / 4.0) as u64;
952            let required_fee = bitcoin::Amount::from_sat(required_fee_sats);
953
954            println!(
955                "Parent weight: {parent_weight}, estimated total: {total_weight}, required fee: {} sats",
956                required_fee.to_sat()
957            );
958
959            let unspent = rpc
960                .list_unspent(
961                    Some(1),
962                    Some(999999999),
963                    Some(&[&fee_payer_address.clone()]),
964                    None,
965                    None,
966                )
967                .await
968                .expect("Failed to list unspent outputs");
969
970            if unspent.is_empty() {
971                let unspent = rpc
972                    .list_unspent(None, None, Some(&[&fee_payer_address.clone()]), None, None)
973                    .await
974                    .expect("Failed to list unspent outputs");
975                if unspent.is_empty() {
976                    println!("No unspent outputs available for fee payment.");
977                    println!("Please send some funds to the fee payer address.");
978                    println!("Fee payer address: {fee_payer_address}");
979                } else {
980                    println!("Unspent outputs: {unspent:?}");
981                    println!("Please wait for them to confirm.");
982                }
983                return;
984            }
985
986            let fee_payer_utxo = unspent
987                .iter()
988                .find(|utxo| utxo.amount > required_fee)
989                .unwrap_or_else(|| {
990                    panic!(
991                        "No UTXO found with enough balance for fee payment, required fee is: {required_fee}"
992                    )
993                });
994
995            // Create child transaction
996            use bitcoin::{transaction::Version, OutPoint, Sequence, TxIn, TxOut};
997
998            let child_input = TxIn {
999                previous_output: OutPoint {
1000                    txid: tx.compute_txid(),
1001                    vout: p2a_vout as u32,
1002                },
1003                script_sig: bitcoin::ScriptBuf::new(),
1004                sequence: Sequence::ENABLE_RBF_NO_LOCKTIME,
1005                witness: bitcoin::Witness::new(),
1006            };
1007
1008            let fee_payer_input = TxIn {
1009                previous_output: OutPoint {
1010                    txid: fee_payer_utxo.txid,
1011                    vout: fee_payer_utxo.vout,
1012                },
1013                script_sig: bitcoin::ScriptBuf::new(),
1014                sequence: Sequence::ENABLE_RBF_NO_LOCKTIME,
1015                witness: bitcoin::Witness::new(),
1016            };
1017
1018            let total_input_value = p2a_txout.value + fee_payer_utxo.amount;
1019            let change_amount = total_input_value
1020                .checked_sub(required_fee)
1021                .expect("Insufficient funds for required fee");
1022
1023            let child_output = TxOut {
1024                value: change_amount,
1025                script_pubkey: fee_payer_address.script_pubkey(),
1026            };
1027
1028            let child_input_utxo = SignRawTransactionInput {
1029                txid: child_input.previous_output.txid,
1030                vout: child_input.previous_output.vout,
1031                script_pub_key: p2a_txout.script_pubkey,
1032                redeem_script: None,
1033                amount: Some(p2a_txout.value),
1034            };
1035
1036            let child_tx = bitcoin::Transaction {
1037                version: Version::non_standard(3),
1038                lock_time: bitcoin::absolute::LockTime::ZERO,
1039                input: vec![child_input, fee_payer_input],
1040                output: vec![child_output],
1041            };
1042
1043            let signed_tx = rpc
1044                .sign_raw_transaction_with_wallet(&child_tx, Some(&[child_input_utxo]), None)
1045                .await
1046                .expect("Failed to sign child transaction");
1047
1048            let signed_child_tx = signed_tx
1049                .transaction()
1050                .expect("Failed to get transaction from sign_raw_transaction_with_wallet");
1051
1052            println!(
1053                "Child transaction signed: {}",
1054                signed_child_tx.compute_txid()
1055            );
1056
1057            // Submit CPFP package
1058            let package = vec![&tx, &signed_child_tx];
1059            println!("Submitting CPFP package");
1060
1061            match rpc
1062                .submit_package(&package, Some(bitcoin::Amount::ZERO), None)
1063                .await
1064            {
1065                Ok(result) => {
1066                    println!("CPFP package submitted successfully");
1067                    println!("Package result: {result:?}");
1068                    let parent_txid = tx.compute_txid();
1069                    println!("Parent transaction TXID: {parent_txid}");
1070                    let child_txid = signed_child_tx.compute_txid();
1071                    println!("Child transaction TXID: {child_txid}");
1072                }
1073                Err(e) => {
1074                    println!("Failed to submit CPFP package: {e}");
1075                    println!("Manual submission options:");
1076                    println!(
1077                        "Parent tx: {}",
1078                        hex::encode(bitcoin::consensus::serialize(&tx))
1079                    );
1080                    println!(
1081                        "Child tx: {}",
1082                        hex::encode(bitcoin::consensus::serialize(&signed_child_tx))
1083                    );
1084                }
1085            }
1086        }
1087    }
1088}
