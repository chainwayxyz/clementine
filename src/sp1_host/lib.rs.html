16/// Proving backend, selected by SP1's standard environment variable.
17#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
18pub enum Backend {
19    Mock,
20    Cpu,
21    Network,
22}
23
24impl Backend {
25    /// Reads the backend from `SP1_PROVER`, which defaults to `network`.
26    pub fn from_env() -> Result<Self> {
27        match std::env::var("SP1_PROVER").as_deref() {
28            Ok("mock") => Ok(Self::Mock),
29            Ok("cpu") => Ok(Self::Cpu),
30            Ok("network") | Err(VarError::NotPresent) => Ok(Self::Network),
31            Ok(other) => Err(eyre!(
32                "Unsupported SP1_PROVER={other}; use mock, cpu or network"
33            )),
34            Err(VarError::NotUnicode(_)) => Err(eyre!("SP1_PROVER is not valid Unicode")),
35        }
36    }
37
38    async fn client(self) -> &'static EnvProver {
39        static MOCK: OnceCell<EnvProver> = OnceCell::const_new();
40        static CPU: OnceCell<EnvProver> = OnceCell::const_new();
41        static NETWORK: OnceCell<EnvProver> = OnceCell::const_new();
42        match self {
43            Self::Mock => {
44                MOCK.get_or_init(|| async {
45                    EnvProver::Mock(ProverClient::builder().mock().build().await)
46                })
47                .await
48            }
49            Self::Cpu => {
50                CPU.get_or_init(|| async {
51                    EnvProver::Cpu(ProverClient::builder().cpu().build().await)
52                })
53                .await
54            }
55            Self::Network => {
56                NETWORK
57                    .get_or_init(|| async {
58                        EnvProver::Network(ProverClient::builder().network().build().await)
59                    })
60                    .await
61            }
62        }
63    }
64}

66/// A cached prover and key for one guest and backend.
67pub struct Host {
68    client: &'static EnvProver,
69    key: EnvProvingKey,
70}
71
72type HostCell = Arc<OnceCell<Arc<Host>>>;
73type HostCache = HashMap<(Backend, [u8; 32]), HostCell>;
74static HOSTS: LazyLock<Mutex<HostCache>> = LazyLock::new(Default::default);
75
76/// Execute without preparing a proving key or checking recursion witnesses.
77pub async fn execute(elf: &[u8], stdin: SP1Stdin) -> Result<Vec<u8>> {
78    run(
79        Backend::Mock.client().await,
80        Elf::from(elf.to_vec()),
81        stdin,
82        false,
83    )
84    .await
85    .map(|(values, _, _)| values)
86}

88/// Verifies a compressed proof of `vk` with public values `public_values`, including the
89/// successful exit code, with SP1's full verifier.
90///
91/// This uses a light prover rather than a backend client: SP1's mock prover accepts every
92/// compressed proof without checking it.
93pub async fn verify_compressed(
94    proof: SP1Proof,
95    public_values: &[u8],
96    vk: SP1VerifyingKey,
97) -> Result<()> {
98    static LIGHT: OnceCell<LightProver> = OnceCell::const_new();
99    eyre::ensure!(
100        matches!(proof, SP1Proof::Compressed(_)),
101        "Expected a compressed SP1 proof"
102    );
103    let prover = LIGHT
104        .get_or_init(|| async { ProverClient::builder().light().build().await })
105        .await;
106    let proof = SP1ProofWithPublicValues::new(
107        proof,
108        SP1PublicValues::from(public_values),
109        prover.version().to_string(),
110    );
111    // The SDK verifies synchronously, so keep the STARK verification off the async workers.
112    tokio::task::spawn_blocking(move || prover.verify(&proof, &vk, None))
113        .await
114        .map_err(|e| eyre!("Compressed SP1 verification task failed: {e}"))?
115        .map_err(|e| eyre!("Compressed SP1 verification failed: {e}"))
116}

118impl Host {
119    /// Concurrent callers share the same setup, without retaining another copy of each ELF.
120    pub async fn get(elf: &[u8], backend: Backend) -> Result<Arc<Self>> {
121        let digest: [u8; 32] = Sha256::digest(elf).into();
122        let cell = HOSTS
123            .lock()
124            .map_err(|_| eyre!("SP1 host cache poisoned"))?
125            .entry((backend, digest))
126            .or_default()
127            .clone();
128        Ok(cell
129            .get_or_try_init(|| async {
130                let client = backend.client().await;
131                let key = client
132                    .setup(Elf::from(elf.to_vec()))
133                    .await
134                    .map_err(|e| eyre!("SP1 setup failed: {e}"))?;
135                Ok::<_, eyre::Report>(Arc::new(Self { client, key }))
136            })
137            .await?
138            .clone())
139    }
140
141    pub fn verifying_key(&self) -> &SP1VerifyingKey {
142        self.key.verifying_key()
143    }
144
145    pub fn program_id(&self) -> Result<[u8; 32]> {
146        hex::decode(self.verifying_key().bytes32().trim_start_matches("0x"))?
147            .try_into()
148            .map_err(|_| eyre!("Invalid SP1 program key"))
149    }
150
151    /// Execute without verifying recursion witnesses, for explicit mock execution only.
152    pub async fn execute(&self, stdin: SP1Stdin) -> Result<Vec<u8>> {
153        self.run(stdin, false).await.map(|(values, _, _)| values)
154    }
155
156    async fn run(&self, stdin: SP1Stdin, preflight: bool) -> Result<(Vec<u8>, u64, Option<u64>)> {
157        run(self.client, self.key.elf().clone(), stdin, preflight).await
158    }
159
160    /// Prove with a caller-selected nonce. BLAKE3 network requests need the SDK 6.10
161    /// compatibility path below until the network accepts its BLAKE3 simulation hint.
162    pub async fn prove(
163        &self,
164        stdin: SP1Stdin,
165        mode: SP1ProofMode,
166        nonce: [u32; 4],
167        blake3: bool,
168    ) -> Result<SP1ProofWithPublicValues> {
169        if matches!(self.client, EnvProver::Cpu(_)) {
170            tracing::warn!(
171                ?mode,
172                "Proving on the local CPU (SP1_PROVER=cpu). Large proofs, such as header chain proofs from genesis, can run out of memory"
173            );
174        }
175        if let (true, EnvProver::Network(network), EnvProvingKey::Network { pk, .. }) =
176            (blake3, self.client, &self.key)
177        {
178            // Keep preflight (including deferred proof checks), omit only its incorrect
179            // hash hint. Like SDK simulation, use the measured cycles and gas as limits.
180            let (expected, cycles, gas) = self.run(stdin.clone(), true).await?;
181            let gas = gas.ok_or_else(|| eyre!("SP1 preflight did not calculate gas"))?;
182            eyre::ensure!(cycles > 0 && gas > 0, "SP1 preflight reported zero usage");
183            let proof = network
184                .prove(pk, stdin)
185                .mode(mode)
186                .with_proof_nonce(nonce)
187                .skip_simulation(true)
188                .cycle_limit(cycles)
189                .gas_limit(gas)
190                .await
191                .map_err(|e| eyre!("SP1 proving failed: {e}"))?;
192            eyre::ensure!(
193                proof.public_values.as_slice() == expected,
194                "SP1 proof public values differ from local preflight"
195            );
196            Ok(proof)
197        } else {
198            self.client
199                .prove(&self.key, stdin)
200                .mode(mode)
201                .with_proof_nonce(nonce)
202                .await
203                .map_err(|e| eyre!("SP1 proving failed: {e}"))
204        }
205    }
206}

208async fn run(
209    client: &EnvProver,
210    elf: Elf,
211    stdin: SP1Stdin,
212    preflight: bool,
213) -> Result<(Vec<u8>, u64, Option<u64>)> {
214    let (stderr_tx, stderr_rx) = tokio::sync::watch::channel(String::new());
215    // Keep SP1's default gas calculation: without it, SDK 6.10 reports exit code 0 and no
216    // cycles even when the guest panics.
217    let (values, report) = client
218        .execute(elf, stdin)
219        .deferred_proof_verification(preflight)
220        .stderr(stderr_tx)
221        .await
222        .map_err(|e| eyre!("SP1 execution failed: {e}: {}", stderr_rx.borrow().as_str()))?;
223    eyre::ensure!(
224        report.exit_code == 0,
225        "SP1 guest exited with code {}: {}",
226        report.exit_code,
227        stderr_rx.borrow().as_str()
228    );
229    Ok((
230        values.to_vec(),
231        report.total_instruction_count(),
232        report.gas(),
233    ))
234}

236#[cfg(test)]
237mod tests {
238    use super::*;
239    use circuits_lib::{
240        common::programs::REGTEST,
241        header_chain::{
242            BlockHeaderCircuitOutput, ChainState, HeaderChainCircuitInput, HeaderChainPrevProofType,
243        },
244    };
245
246    #[tokio::test]
247    async fn shared_setup_and_preflight_preserve_execution_checks() {
248        let elf = include_bytes!("../../sp1-circuits/elfs/regtest-header-chain.elf");
249        let (first, second) =
250            tokio::join!(Host::get(elf, Backend::Mock), Host::get(elf, Backend::Mock));
251        let first = first.unwrap();
252        assert!(Arc::ptr_eq(&first, &second.unwrap()));
253        assert_eq!(first.program_id().unwrap(), REGTEST.header_chain.groth16);
254        let input = HeaderChainCircuitInput {
255            method_id: REGTEST.header_chain.recursion,
256            prev_proof: HeaderChainPrevProofType::GenesisBlock(ChainState::genesis_state()),
257            block_headers: vec![],
258        };
259        let mut stdin = SP1Stdin::new();
260        stdin.write_vec(borsh::to_vec(&input).unwrap());
261        let (values, cycles, gas) = first.run(stdin.clone(), true).await.unwrap();
262        let output: BlockHeaderCircuitOutput = borsh::from_slice(&values).unwrap();
263        assert_eq!(output.method_id, input.method_id);
264        assert!(cycles > 0 && gas.unwrap() > 0);
265        assert_eq!(values, first.execute(stdin.clone()).await.unwrap());
266
267        // The mock prover's compressed proof is a placeholder that real verification rejects.
268        let mock = first
269            .prove(stdin.clone(), SP1ProofMode::Compressed, [0; 4], false)
270            .await
271            .unwrap();
272        let vk = first.verifying_key().clone();
273        let error = verify_compressed(mock.proof, &values, vk)
274            .await
275            .unwrap_err();
276        assert!(error
277            .to_string()
278            .starts_with("Compressed SP1 verification failed"));
279
280        assert_eq!(values, execute(elf, stdin).await.unwrap());
281        let mut invalid = SP1Stdin::new();
282        invalid.write_vec(vec![0xff]);
283        assert!(first.run(invalid, true).await.is_err());
284    }
